## BASIC AXES SETUP
```python
from manim import *
import numpy as np

# Standard 2D Axes
axes = Axes(
    x_range=[-5, 5, 1],      # [min, max, step]
    y_range=[-3, 3, 1],
    x_length=10,              # Screen width
    y_length=6,               # Screen height
    axis_config={
        "color": WHITE,
        "include_tip": True,
        "include_numbers": True
    }
)
self.play(Create(axes), run_time=1.5)

# Axes with labels
x_label = axes.get_x_axis_label(r"x", direction=RIGHT)
y_label = axes.get_y_axis_label(r"y", direction=UP)
self.play(Write(x_label), Write(y_label), run_time=0.8)

# NumberPlane (with grid)
plane = NumberPlane(
    x_range=[-5, 5, 1],
    y_range=[-3, 3, 1],
    background_line_style={
        "stroke_color": BLUE,
        "stroke_opacity": 0.3
    }
)
self.play(Create(plane), run_time=2.0)
```

## PLOTTING FUNCTIONS
```python
# Basic function plot
axes = Axes(x_range=[-3, 3], y_range=[-1, 9])
graph = axes.plot(lambda x: x**2, color=BLUE)

self.play(Create(axes), run_time=1.0)
self.play(Create(graph), run_time=2.0)

# Plot with label
label = axes.get_graph_label(graph, label=r"y = x^2", direction=UR)
self.play(Write(label), run_time=0.8)

# Multiple functions on same axes
axes = Axes(x_range=[-4, 4], y_range=[-2, 2])
sin_graph = axes.plot(lambda x: np.sin(x), color=BLUE)
cos_graph = axes.plot(lambda x: np.cos(x), color=RED)

self.play(Create(axes))
self.play(Create(sin_graph), run_time=1.5)
self.play(Create(cos_graph), run_time=1.5)

# Different domain for each function
graph1 = axes.plot(lambda x: x**2, x_range=[-2, 2], color=BLUE)
graph2 = axes.plot(lambda x: x**3, x_range=[-1.5, 1.5], color=GREEN)
```

## PARAMETRIC CURVES
```python
# Parametric curve (x(t), y(t))
axes = Axes(x_range=[-2, 2], y_range=[-2, 2])

```

## AREA UNDER CURVE (Riemann Sums)
```python
# Setup
axes = Axes(x_range=[0, 5, 1], y_range=[0, 30, 5])
graph = axes.plot(lambda x: x**2, x_range=[0, 5], color=BLUE)

self.play(Create(axes), Create(graph), run_time=1.5)

# Riemann rectangles (approximation)
riemann_rects = axes.get_riemann_rectangles(
    graph,
    x_range=[0, 4],
    dx=0.5,                    # Rectangle width
    input_sample_type="left",  # left, right, or center
    fill_opacity=0.5,
    color=GREEN
)
self.play(Create(riemann_rects), run_time=2.0)

# Animate refinement
for dx in [0.5, 0.25, 0.1]:
    new_rects = axes.get_riemann_rectangles(
        graph, x_range=[0, 4], dx=dx,
        fill_opacity=0.5, color=GREEN
    )
    self.play(Transform(riemann_rects, new_rects), run_time=1.0)
    self.wait(0.5)

# Area fill (exact integral)
area = axes.get_area(graph, x_range=[0, 4], color=GREEN, opacity=0.5)
self.play(Create(area), run_time=1.5)
```

## TANGENT LINES AND DERIVATIVES
```python
# Function and tangent line
axes = Axes(x_range=[-3, 3], y_range=[-1, 9])
graph = axes.plot(lambda x: x**2, color=BLUE)

self.play(Create(axes), Create(graph))

# Tangent at specific point
x_val = 1
tangent_line = axes.get_secant_slope_group(
    x=x_val,
    graph=graph,
    dx=0.01,
    secant_line_color=YELLOW,
    secant_line_length=4
)
self.play(Create(tangent_line), run_time=1.0)

# Moving tangent line along curve
def get_tangent(x):
    return axes.get_secant_slope_group(
        x=x, graph=graph, dx=0.01,
        secant_line_color=YELLOW, secant_line_length=3
    )

tangent = get_tangent(-2)
self.play(Create(tangent))

for x in np.linspace(-2, 2, 20):
    new_tangent = get_tangent(x)
    self.play(Transform(tangent, new_tangent), run_time=0.1)
```

## DOTS AND MARKERS ON GRAPHS
```python
# Point on graph
axes = Axes(x_range=[-3, 3], y_range=[-1, 9])
graph = axes.plot(lambda x: x**2, color=BLUE)

# Create dot at specific point
x_val = 2
y_val = x_val**2
point = Dot(axes.c2p(x_val, y_val), color=RED, radius=0.1)

self.play(Create(axes), Create(graph))
self.play(Create(point), run_time=0.5)

# Label the point
coord_label = MathTex(f"({x_val}, {y_val})", font_size=28, color=YELLOW)
coord_label.next_to(point, UR, buff=0.2)
self.play(Write(coord_label), run_time=0.8)

# Animated dot along curve
moving_dot = Dot(color=YELLOW, radius=0.08)
moving_dot.move_to(axes.c2p(-3, 9))

self.play(
    MoveAlongPath(moving_dot, graph),
    run_time=4,
    rate_func=linear
)
```

## COMPLETE EXAMPLE: Parabola with Roots
```python
from manim import *

class ParabolaRoots(Scene):
    def construct(self):
        # Title
        title = Text("Finding Roots Graphically", font_size=40, color=YELLOW)
        title.to_edge(UP)
        self.play(Write(title), run_time=1.0)
        
        # Create axes
        axes = Axes(
            x_range=[-5, 3, 1],
            y_range=[-5, 10, 2],
            x_length=8,
            y_length=5,
            axis_config={"color": WHITE, "include_numbers": True}
        )
        axes.shift(DOWN * 0.5)
        
        # Function: y = x² + 2x - 3 = (x+3)(x-1)
        graph = axes.plot(
            lambda x: x**2 + 2*x - 3,
            x_range=[-4.5, 2.5],
            color=BLUE
        )
        
        # Label
        label = MathTex(r"y = x^2 + 2x - 3", font_size=32, color=BLUE)
        label.next_to(graph, UR, buff=0.5)
        
        self.play(Create(axes), run_time=1.5)
        self.play(Create(graph), Write(label), run_time=2.0)
        
        # Mark the roots
        root1 = Dot(axes.c2p(-3, 0), color=GREEN, radius=0.12)
        root2 = Dot(axes.c2p(1, 0), color=GREEN, radius=0.12)
        
        root1_label = MathTex("x = -3", font_size=28, color=GREEN)
        root2_label = MathTex("x = 1", font_size=28, color=GREEN)
        root1_label.next_to(root1, DOWN)
        root2_label.next_to(root2, DOWN)
        
        self.play(Create(root1), Write(root1_label), run_time=0.8)
        self.play(Create(root2), Write(root2_label), run_time=0.8)
        
        # Highlight x-axis (where y = 0)
        x_axis_highlight = axes.get_horizontal_line(
            axes.c2p(0, 0), color=YELLOW, stroke_width=3
        )
        
        # Factored form
        factored = MathTex(
            r"(x + 3)(x - 1) = 0",
            font_size=36,
            color=YELLOW
        )
        factored.to_edge(DOWN)
        
        self.play(Write(factored), run_time=1.5)
        self.play(
            Indicate(root1, color=GREEN),
            Indicate(root2, color=GREEN),
            run_time=0.8
        )
        
        self.wait(2.0)
```

## COORDINATE TRANSFORMS (c2p and p2c)
```python
# c2p = coordinates to point (math coords → screen position)
# p2c = point to coordinates (screen position → math coords)

axes = Axes(x_range=[-5, 5], y_range=[-3, 3])

# Convert math coordinates to screen position
screen_point = axes.c2p(2, 1)  # Where (2,1) is on screen
dot = Dot(screen_point, color=RED)

# Get coordinates from screen position
math_coords = axes.p2c(screen_point)  # Returns [2, 1, 0]

# Use c2p for labels
label = MathTex("(2, 1)", font_size=24)
label.move_to(axes.c2p(2, 1) + UP * 0.3)
```

## IMPLICIT CURVES
```python
# For curves defined implicitly like x² + y² = 1
axes = Axes(x_range=[-2, 2], y_range=[-2, 2])

# Create implicit curve
circle = ImplicitFunction(
    lambda x, y: x**2 + y**2 - 1,  # = 0
    color=BLUE
)
self.play(Create(axes), Create(circle), run_time=2.0)

# Ellipse
ellipse = ImplicitFunction(
    lambda x, y: x**2/4 + y**2 - 1,
    color=YELLOW
)
self.play(Create(ellipse), run_time=1.5)
```
