## 1. Reflection in Axis Lines

```python
from manim import *
import numpy as np

class ReflectionInAxes(Scene):
    def construct(self):
        title = Text("Reflection", font_size=40, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Coordinate axes
        axes = Axes(
            x_range=[-5, 5, 1],
            y_range=[-4, 4, 1],
            x_length=8,
            y_length=6,
            axis_config={"include_tip": True, "numbers_to_include": [-4, -2, 2, 4]}
        )
        self.play(Create(axes))
        
        # Y-axis as mirror line
        mirror = DashedLine(UP * 3, DOWN * 3, color=YELLOW, stroke_width=3)
        mirror_label = Text("y-axis", font_size=20, color=YELLOW)
        mirror_label.next_to(mirror, UP)
        self.play(Create(mirror), Write(mirror_label))
        
        # Original shape (triangle)
        original = Polygon(
            axes.c2p(1, 1), axes.c2p(3, 1), axes.c2p(2, 3),
            color=RED, fill_opacity=0.5
        )
        original_label = Text("Original", font_size=18, color=RED)
        original_label.next_to(original, DOWN, buff=0.1)
        
        self.play(Create(original), Write(original_label))
        
        # Reflected shape
        reflected = Polygon(
            axes.c2p(-1, 1), axes.c2p(-3, 1), axes.c2p(-2, 3),
            color=GREEN, fill_opacity=0.5
        )
        reflected_label = Text("Image", font_size=18, color=GREEN)
        reflected_label.next_to(reflected, DOWN, buff=0.1)
        
        self.play(ReplacementTransform(original.copy(), reflected), Write(reflected_label))
        
        # Rule
        rule = MathTex(r"(x, y) \rightarrow (-x, y)", font_size=28, color=YELLOW)
        rule.to_edge(DOWN, buff=0.5)
        self.play(Write(rule))
        
        self.wait(2)
```

## 2. Reflection in Line y = x

```python
from manim import *
import numpy as np

class ReflectionInYEqualsX(Scene):
    def construct(self):
        title = Text("Reflection in y = x", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Coordinate axes
        axes = Axes(
            x_range=[-5, 5, 1],
            y_range=[-4, 4, 1],
            x_length=8,
            y_length=6,
            axis_config={"include_tip": True}
        )
        self.play(Create(axes))
        
        # Line y = x
        mirror = Line(axes.c2p(-4, -4), axes.c2p(4, 4), color=YELLOW, stroke_width=3)
        mirror_label = MathTex("y = x", font_size=24, color=YELLOW)
        mirror_label.next_to(axes.c2p(3, 3), RIGHT)
        
        self.play(Create(mirror), Write(mirror_label))
        
        # Original point
        point_A = Dot(axes.c2p(1, 3), color=RED)
        A_label = MathTex("A(1, 3)", font_size=24, color=RED)
        A_label.next_to(point_A, RIGHT)
        
        self.play(Create(point_A), Write(A_label))
        
        # Reflected point
        point_A_prime = Dot(axes.c2p(3, 1), color=GREEN)
        A_prime_label = MathTex("A'(3, 1)", font_size=24, color=GREEN)
        A_prime_label.next_to(point_A_prime, RIGHT)
        
        # Show the transformation
        self.play(
            ReplacementTransform(point_A.copy(), point_A_prime),
            Write(A_prime_label)
        )
        
        # Connecting line (perpendicular to y=x)
        connector = DashedLine(axes.c2p(1, 3), axes.c2p(3, 1), color=WHITE)
        self.play(Create(connector))
        
        # Rule
        rule = MathTex(r"(x, y) \rightarrow (y, x)", font_size=28, color=YELLOW)
        rule.to_edge(DOWN, buff=0.5)
        box = SurroundingRectangle(rule, color=YELLOW)
        
        self.play(Write(rule), Create(box))
        
        # More examples
        points = [(2, -1), (-3, 2), (-2, -3)]
        colors = [PURPLE, ORANGE, TEAL]
        
        for (x, y), col in zip(points, colors):
            orig = Dot(axes.c2p(x, y), color=col)
            refl = Dot(axes.c2p(y, x), color=col)
            self.play(Create(orig), run_time=0.3)
            self.play(ReplacementTransform(orig.copy(), refl), run_time=0.5)
        
        self.wait(2)
```

## 3. Rotation about Origin

```python
from manim import *
import numpy as np

class RotationAboutOrigin(Scene):
    def construct(self):
        title = Text("Rotation about Origin", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Coordinate axes
        axes = Axes(
            x_range=[-5, 5, 1],
            y_range=[-4, 4, 1],
            x_length=8,
            y_length=6,
            axis_config={"include_tip": True}
        )
        self.play(Create(axes))
        
        # Center of rotation
        center = Dot(axes.c2p(0, 0), color=YELLOW)
        center_label = Text("O", font_size=20, color=YELLOW)
        center_label.next_to(center, DOWN + LEFT, buff=0.1)
        
        self.play(Create(center), Write(center_label))
        
        # Original shape
        original = Polygon(
            axes.c2p(1, 1), axes.c2p(3, 1), axes.c2p(3, 2), axes.c2p(1, 2),
            color=RED, fill_opacity=0.5
        )
        self.play(Create(original))
        
        # Show 90° counterclockwise rotation
        rotation_info = Text("90° counterclockwise", font_size=24, color=GREEN)
        rotation_info.to_edge(RIGHT).shift(UP * 2)
        
        self.play(Write(rotation_info))
        
        # Rotated shape
        rotated = Polygon(
            axes.c2p(-1, 1), axes.c2p(-1, 3), axes.c2p(-2, 3), axes.c2p(-2, 1),
            color=GREEN, fill_opacity=0.5
        )
        
        # Animate rotation
        self.play(Rotate(original.copy(), angle=PI/2, about_point=axes.c2p(0, 0)))
        self.play(Create(rotated))
        
        # Rule
        rule = MathTex(r"90° \text{ CCW}: (x, y) \rightarrow (-y, x)", font_size=26, color=GREEN)
        rule.to_edge(DOWN, buff=0.8)
        
        self.play(Write(rule))
        
        # Show 180° rotation
        self.play(FadeOut(rotation_info), FadeOut(rotated), FadeOut(rule))
        
        rotation_info2 = Text("180° rotation", font_size=24, color=PURPLE)
        rotation_info2.to_edge(RIGHT).shift(UP * 2)
        self.play(Write(rotation_info2))
        
        rotated_180 = Polygon(
            axes.c2p(-1, -1), axes.c2p(-3, -1), axes.c2p(-3, -2), axes.c2p(-1, -2),
            color=PURPLE, fill_opacity=0.5
        )
        
        self.play(Rotate(original.copy(), angle=PI, about_point=axes.c2p(0, 0)))
        self.play(Create(rotated_180))
        
        rule2 = MathTex(r"180°: (x, y) \rightarrow (-x, -y)", font_size=26, color=PURPLE)
        rule2.to_edge(DOWN, buff=0.8)
        
        self.play(Write(rule2))
        
        self.wait(2)
```

## 4. Rotation about Any Point

```python
from manim import *
import numpy as np

class RotationAboutPoint(Scene):
    def construct(self):
        title = Text("Rotation about a Point", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Coordinate axes
        axes = Axes(
            x_range=[-2, 6, 1],
            y_range=[-2, 5, 1],
            x_length=8,
            y_length=6,
            axis_config={"include_tip": True}
        )
        self.play(Create(axes))
        
        # Center of rotation
        center = Dot(axes.c2p(2, 1), color=YELLOW)
        center_label = MathTex("(2, 1)", font_size=20, color=YELLOW)
        center_label.next_to(center, DOWN, buff=0.1)
        
        self.play(Create(center), Write(center_label))
        
        # Original triangle
        original = Polygon(
            axes.c2p(3, 2), axes.c2p(5, 2), axes.c2p(4, 4),
            color=RED, fill_opacity=0.5
        )
        orig_label = Text("A", font_size=20, color=RED)
        orig_label.move_to(axes.c2p(4, 2.5))
        
        self.play(Create(original), Write(orig_label))
        
        # Rotation angle arc
        angle_arc = Arc(
            radius=0.5,
            start_angle=np.arctan2(1, 1),
            angle=PI/2,
            arc_center=axes.c2p(2, 1),
            color=GREEN
        )
        angle_label = MathTex("90°", font_size=20, color=GREEN)
        angle_label.next_to(angle_arc, RIGHT, buff=0.1)
        
        self.play(Create(angle_arc), Write(angle_label))
        
        # Show rotation steps
        step1 = Text("1. Translate center to origin", font_size=18, color=WHITE)
        step2 = Text("2. Rotate", font_size=18, color=WHITE)
        step3 = Text("3. Translate back", font_size=18, color=WHITE)
        
        steps = VGroup(step1, step2, step3).arrange(DOWN, aligned_edge=LEFT)
        steps.to_edge(LEFT).shift(DOWN)
        
        self.play(Write(steps))
        
        # Rotated triangle (90° counterclockwise about (2,1))
        # Original vertex (3,2) relative to (2,1) is (1,1)
        # After 90° CCW: (-1, 1), then translate back: (1, 2)
        rotated = Polygon(
            axes.c2p(1, 2), axes.c2p(1, 4), axes.c2p(-1, 3),
            color=GREEN, fill_opacity=0.5
        )
        
        # Animate the rotation
        temp_shape = original.copy()
        self.play(
            Rotate(temp_shape, angle=PI/2, about_point=axes.c2p(2, 1)),
            run_time=1.5
        )
        self.remove(temp_shape)
        
        rotated_label = Text("A'", font_size=20, color=GREEN)
        rotated_label.move_to(axes.c2p(0, 3))
        
        self.play(Create(rotated), Write(rotated_label))
        
        self.wait(2)
```

## 6. Enlargement with Positive Scale Factor

```python
from manim import *
import numpy as np

class EnlargementPositive(Scene):
    def construct(self):
        title = Text("Enlargement (k > 1)", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Coordinate axes
        axes = Axes(
            x_range=[-2, 10, 1],
            y_range=[-2, 8, 1],
            x_length=8,
            y_length=6,
            axis_config={"include_tip": True}
        )
        self.play(Create(axes))
        
        # Center of enlargement
        center = Dot(axes.c2p(1, 1), color=YELLOW)
        center_label = MathTex("C(1, 1)", font_size=20, color=YELLOW)
        center_label.next_to(center, DOWN + LEFT, buff=0.1)
        
        self.play(Create(center), Write(center_label))
        
        # Original shape
        original = Polygon(
            axes.c2p(2, 2), axes.c2p(4, 2), axes.c2p(3, 4),
            color=RED, fill_opacity=0.5
        )
        self.play(Create(original))
        
        # Scale factor
        sf_label = MathTex(r"\text{Scale factor } k = 2", font_size=28, color=GREEN)
        sf_label.to_edge(RIGHT).shift(UP * 2)
        
        self.play(Write(sf_label))
        
        # Draw rays from center through vertices
        rays = VGroup()
        vertices = [axes.c2p(2, 2), axes.c2p(4, 2), axes.c2p(3, 4)]
        for v in vertices:
            direction = v - axes.c2p(1, 1)
            ray = DashedLine(axes.c2p(1, 1), axes.c2p(1, 1) + direction * 2.5, color=GRAY)
            rays.add(ray)
        
        self.play(Create(rays))
        
        # Enlarged shape (scale factor 2 from center (1,1))
        # (2,2) -> (1,1) + 2*((2,2)-(1,1)) = (1,1) + 2*(1,1) = (3,3)
        # (4,2) -> (1,1) + 2*((4,2)-(1,1)) = (1,1) + 2*(3,1) = (7,3)
        # (3,4) -> (1,1) + 2*((3,4)-(1,1)) = (1,1) + 2*(2,3) = (5,7)
        enlarged = Polygon(
            axes.c2p(3, 3), axes.c2p(7, 3), axes.c2p(5, 7),
            color=GREEN, fill_opacity=0.5
        )
        
        self.play(Create(enlarged))
        
        # Labels
        original_label = Text("Original", font_size=18, color=RED)
        original_label.next_to(original, DOWN)
        image_label = Text("Image", font_size=18, color=GREEN)
        image_label.next_to(enlarged, RIGHT)
        
        self.play(Write(original_label), Write(image_label))
        
        # Note about area
        area_note = MathTex(r"\text{Area factor} = k^2 = 4", font_size=24, color=ORANGE)
        area_note.to_edge(DOWN, buff=0.5)
        
        self.play(Write(area_note))
        
        self.wait(2)
```

## 9. Combined Transformations

```python
from manim import *
import numpy as np

class CombinedTransformations(Scene):
    def construct(self):
        title = Text("Combined Transformations", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Coordinate axes
        axes = Axes(
            x_range=[-5, 5, 1],
            y_range=[-4, 4, 1],
            x_length=8,
            y_length=6,
            axis_config={"include_tip": True}
        )
        self.play(Create(axes))
        
        # Original shape
        original = Polygon(
            axes.c2p(1, 1), axes.c2p(3, 1), axes.c2p(2, 2),
            color=RED, fill_opacity=0.5
        )
        orig_label = Text("A", font_size=20, color=RED)
        orig_label.move_to(axes.c2p(2, 1.3))
        
        self.play(Create(original), Write(orig_label))
        
        # Step 1: Reflection in y-axis
        step1 = Text("Step 1: Reflect in y-axis", font_size=20, color=YELLOW)
        step1.to_edge(LEFT).shift(UP * 1)
        
        self.play(Write(step1))
        
        reflected = Polygon(
            axes.c2p(-1, 1), axes.c2p(-3, 1), axes.c2p(-2, 2),
            color=YELLOW, fill_opacity=0.5
        )
        refl_label = Text("B", font_size=20, color=YELLOW)
        refl_label.move_to(axes.c2p(-2, 1.3))
        
        self.play(
            ReplacementTransform(original.copy(), reflected),
            Write(refl_label)
        )
        
        # Step 2: Rotate 90° clockwise about origin
        step2 = Text("Step 2: Rotate 90° CW about O", font_size=20, color=GREEN)
        step2.next_to(step1, DOWN, aligned_edge=LEFT)
        
        self.play(Write(step2))
        
        # 90° CW: (x,y) -> (y, -x)
        # (-1,1) -> (1, 1)
        # (-3,1) -> (1, 3)
        # (-2,2) -> (2, 2)
        rotated = Polygon(
            axes.c2p(1, 1), axes.c2p(1, 3), axes.c2p(2, 2),
            color=GREEN, fill_opacity=0.5
        )
        rot_label = Text("C", font_size=20, color=GREEN)
        rot_label.move_to(axes.c2p(1.3, 2))
        
        self.play(
            Rotate(reflected.copy(), angle=-PI/2, about_point=axes.c2p(0, 0)),
            run_time=1
        )
        self.play(Create(rotated), Write(rot_label))
        
        # Final result description
        result = VGroup(
            Text("Single transformation:", font_size=18, color=ORANGE),
            Text("Reflect in y = x", font_size=18, color=ORANGE)
        ).arrange(DOWN, buff=0.1)
        result.to_edge(DOWN, buff=0.5)
        box = SurroundingRectangle(result, color=ORANGE)
        
        self.play(Write(result), Create(box))
        
        self.wait(2)
```

## 11. Inverse Transformations

```python
from manim import *
import numpy as np

class InverseTransformations(Scene):
    def construct(self):
        title = Text("Inverse Transformations", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        subtitle = Text("Transformations that 'undo' each other", font_size=22, color=GRAY)
        subtitle.next_to(title, DOWN, buff=0.2)
        self.play(Write(subtitle))
        
        # Table of inverse transformations
        table_data = [
            ("Transformation", "Inverse", YELLOW),
            ("Reflection in line L", "Reflection in line L", WHITE),
            ("Rotation θ° about P", "Rotation -θ° about P", WHITE),
            ("Translation (a, b)", "Translation (-a, -b)", WHITE),
            ("Enlargement k, center C", "Enlargement 1/k, center C", WHITE),
        ]
        
        rows = VGroup()
        for i, (col1, col2, color) in enumerate(table_data):
            row = VGroup(
                Text(col1, font_size=20, color=color),
                Text(col2, font_size=20, color=color)
            )
            row[0].move_to(LEFT * 2.5)
            row[1].move_to(RIGHT * 2.5)
            row.shift(UP * 1 + DOWN * i * 0.7)
            rows.add(row)
        
        # Add arrow between columns
        arrow_col = VGroup()
        for i in range(1, 5):
            arrow = Arrow(LEFT * 0.5, RIGHT * 0.5, color=GREEN)
            arrow.move_to(UP * 1 + DOWN * i * 0.7)
            arrow_col.add(arrow)
        
        self.play(Write(rows), run_time=2)
        self.play(Create(arrow_col))
        
        # Note
        note = VGroup(
            Text("Note: Reflection is self-inverse", font_size=20, color=ORANGE),
            Text("(reflecting twice returns to original)", font_size=18, color=GRAY)
        ).arrange(DOWN, buff=0.1)
        note.to_edge(DOWN, buff=0.5)
        
        self.play(Write(note))
        
        self.wait(2)
```
