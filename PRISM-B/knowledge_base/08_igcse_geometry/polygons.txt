## 1. Naming Regular Polygons

```python
from manim import *
import numpy as np

class NamingRegularPolygons(Scene):
    def construct(self):
        title = Text("Regular Polygons", font_size=40, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Create polygons with names
        polygons_data = [
            (3, "Triangle", RED),
            (4, "Square", GREEN),
            (5, "Pentagon", YELLOW),
            (6, "Hexagon", ORANGE),
            (8, "Octagon", PURPLE),
            (10, "Decagon", TEAL)
        ]
        
        # First row
        for i, (sides, name, color) in enumerate(polygons_data[:3]):
            polygon = RegularPolygon(n=sides, color=color, fill_opacity=0.3)
            polygon.scale(0.7)
            polygon.shift(LEFT * 4 + RIGHT * i * 4 + UP * 0.5)
            
            label = Text(name, font_size=24, color=color)
            label.next_to(polygon, DOWN)
            
            sides_label = Text(f"{sides} sides", font_size=18, color=WHITE)
            sides_label.next_to(label, DOWN, buff=0.1)
            
            self.play(Create(polygon), Write(label), Write(sides_label), run_time=0.8)
        
        # Second row
        for i, (sides, name, color) in enumerate(polygons_data[3:]):
            polygon = RegularPolygon(n=sides, color=color, fill_opacity=0.3)
            polygon.scale(0.7)
            polygon.shift(LEFT * 4 + RIGHT * i * 4 + DOWN * 2.5)
            
            label = Text(name, font_size=24, color=color)
            label.next_to(polygon, DOWN)
            
            sides_label = Text(f"{sides} sides", font_size=18, color=WHITE)
            sides_label.next_to(label, DOWN, buff=0.1)
            
            self.play(Create(polygon), Write(label), Write(sides_label), run_time=0.8)
        
        self.wait(2)
```

```python
from manim import *

        self.play(Write(formula))

        pentagon = RegularPolygon(n=5, color=YELLOW, fill_opacity=0.2)

        diagonals = VGroup()
        for i in [2, 3]:

            diagonals.add(line)
        
        self.play(Create(diagonals))
        
        # Label triangles
        triangle_label = Text("3 triangles", font_size=24, color=RED)
        triangle_label.next_to(pentagon, DOWN)
        self.play(Write(triangle_label))
        
        # Calculation
        calc = VGroup(
            MathTex(r"n = 5 \text{ (pentagon)}", font_size=28),
            MathTex(r"\text{Sum} = (5 - 2) \times 180°", font_size=28),
            MathTex(r"= 3 \times 180°", font_size=28),
            MathTex(r"= 540°", font_size=28, color=GREEN)
```

## 3. Exterior Angles of Polygons

```python
from manim import *
import numpy as np

class ExteriorAngleSum(Scene):
    def construct(self):
        title = Text("Exterior Angles of Polygons", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Key fact
        fact = MathTex(r"\text{Sum of exterior angles} = 360°", font_size=32, color=GREEN)
        fact.next_to(title, DOWN, buff=0.3)
        box = SurroundingRectangle(fact, color=GREEN)
        self.play(Write(fact), Create(box))
        
        # Hexagon with exterior angles
        hexagon = RegularPolygon(n=6, color=YELLOW)
        hexagon.scale(1.5)
        hexagon.shift(DOWN * 1)
        self.play(Create(hexagon))
        
        vertices = hexagon.get_vertices()
        
        # Draw exterior angles
        exterior_angles = VGroup()
        angle_labels = VGroup()
        
        for i in range(6):
            # Extend side outward
            v1 = vertices[i]
            v2 = vertices[(i + 1) % 6]
            direction = v2 - v1
            direction = direction / np.linalg.norm(direction)
            
            extension = Line(v2, v2 + direction * 0.8, color=RED)
            exterior_angles.add(extension)
            
            # Angle arc
            angle_arc = Arc(
                radius=0.3,
                start_angle=np.arctan2((v2 - v1)[1], (v2 - v1)[0]),
                angle=PI/3,
                arc_center=v2,
                color=RED
            )
            exterior_angles.add(angle_arc)
        
        self.play(Create(exterior_angles))
        
        # Calculation
        calc = VGroup(
            MathTex(r"\text{Regular hexagon: } n = 6", font_size=26),
            MathTex(r"\text{Each exterior angle} = \frac{360°}{6}", font_size=26),
            MathTex(r"= 60°", font_size=26, color=RED)
        ).arrange(DOWN, aligned_edge=LEFT)
        calc.to_edge(RIGHT).shift(DOWN * 0.5)
        
        for line in calc:
            self.play(Write(line), run_time=0.6)
        
        # Interior + Exterior = 180
        relation = MathTex(r"\text{Interior} + \text{Exterior} = 180°", font_size=28, color=ORANGE)
        relation.next_to(calc, DOWN, buff=0.5)
        self.play(Write(relation))
        
        self.wait(2)
```

## 5. Tessellations

```python
from manim import *
import numpy as np

class Tessellations(Scene):
    def construct(self):
        title = Text("Tessellations", font_size=40, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        definition = Text("Shapes that tile with no gaps or overlaps", font_size=24, color=GRAY)
        definition.next_to(title, DOWN, buff=0.3)
        self.play(Write(definition))
        
        # Triangle tessellation
        tri_group = VGroup()
        for i in range(-2, 3):
            for j in range(-1, 2):
                tri = RegularPolygon(n=3, color=RED, fill_opacity=0.3)
                tri.scale(0.4)
                x_offset = i * 0.7 + (0.35 if j % 2 else 0)
                tri.shift(LEFT * 4 + RIGHT * x_offset + UP * j * 0.6 + DOWN * 1)
                if (i + j) % 2 == 0:
                    tri.rotate(PI)
                tri_group.add(tri)
        
        tri_label = Text("Triangles", font_size=20, color=RED)
        tri_label.next_to(tri_group, DOWN)
        
        self.play(Create(tri_group), Write(tri_label))
        
        # Square tessellation
        sq_group = VGroup()
        for i in range(-1, 2):
            for j in range(-1, 2):
                sq = Square(side_length=0.6, color=GREEN, fill_opacity=0.3)
                sq.shift(RIGHT * i * 0.6 + UP * j * 0.6 + DOWN * 1)
                sq_group.add(sq)
        
        sq_label = Text("Squares", font_size=20, color=GREEN)
        sq_label.next_to(sq_group, DOWN)
        
        self.play(Create(sq_group), Write(sq_label))
        
        # Hexagon tessellation
        hex_group = VGroup()
        hex_size = 0.35
        for i in range(-1, 2):
            for j in range(-1, 2):
                hex = RegularPolygon(n=6, color=YELLOW, fill_opacity=0.3)
                hex.scale(hex_size)
                x_offset = i * hex_size * 1.8 + (0.9 * hex_size if j % 2 else 0)
                hex.shift(RIGHT * 4 + RIGHT * x_offset + UP * j * hex_size * 1.5 + DOWN * 1)
                hex_group.add(hex)
        
        hex_label = Text("Hexagons", font_size=20, color=YELLOW)
        hex_label.next_to(hex_group, DOWN)
        
        self.play(Create(hex_group), Write(hex_label))
        
        # Note about angles
        note = Text("Interior angle must divide 360° evenly", font_size=22, color=ORANGE)
        note.to_edge(DOWN, buff=0.5)
        box = SurroundingRectangle(note, color=ORANGE)
        self.play(Write(note), Create(box))
        
        self.wait(2)
```

## 9. Constructing Shapes with Compass and Ruler

```python
from manim import *
import numpy as np

class CompassConstruction(Scene):
    def construct(self):
        title = Text("Compass Constructions", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        subtitle = Text("Constructing an Equilateral Triangle", font_size=24, color=GRAY)
        subtitle.next_to(title, DOWN, buff=0.2)
        self.play(Write(subtitle))
        
        # Step 1: Draw base
        step1 = Text("Step 1: Draw base AB", font_size=20, color=WHITE)
        step1.to_edge(LEFT).shift(UP * 1)
        
        A = LEFT * 2 + DOWN * 1
        B = RIGHT * 2 + DOWN * 1
        base = Line(A, B, color=WHITE)
        
        A_label = MathTex("A", font_size=24).next_to(A, DOWN + LEFT, buff=0.1)
        B_label = MathTex("B", font_size=24).next_to(B, DOWN + RIGHT, buff=0.1)
        
        self.play(Write(step1))
        self.play(Create(base), Write(A_label), Write(B_label))
        
        # Step 2: Arc from A
        step2 = Text("Step 2: Arc from A", font_size=20, color=WHITE)
        step2.next_to(step1, DOWN, buff=0.3)
        
        radius = np.linalg.norm(B - A)
        arc1 = Arc(radius=radius, start_angle=0, angle=PI/3, arc_center=A, color=RED)
        
        self.play(Write(step2))
        self.play(Create(arc1))
        
        # Step 3: Arc from B
        step3 = Text("Step 3: Arc from B", font_size=20, color=WHITE)
        step3.next_to(step2, DOWN, buff=0.3)
        
        arc2 = Arc(radius=radius, start_angle=2*PI/3, angle=PI/3, arc_center=B, color=GREEN)
        
        self.play(Write(step3))
        self.play(Create(arc2))
        
        # Step 4: Mark intersection C
        step4 = Text("Step 4: Mark point C", font_size=20, color=WHITE)
        step4.next_to(step3, DOWN, buff=0.3)
        
        # Calculate C
        C = (A + B) / 2 + UP * radius * np.sin(PI/3)
        C_dot = Dot(C, color=YELLOW)
        C_label = MathTex("C", font_size=24, color=YELLOW).next_to(C, UP, buff=0.1)
        
        self.play(Write(step4))
        self.play(Create(C_dot), Write(C_label))
        
        # Step 5: Connect to form triangle
        step5 = Text("Step 5: Join A-C and B-C", font_size=20, color=WHITE)
        step5.next_to(step4, DOWN, buff=0.3)
        
        AC = Line(A, C, color=YELLOW)
        BC = Line(B, C, color=YELLOW)
        
        self.play(Write(step5))
        self.play(Create(AC), Create(BC))
        
        # Final result
        result = Text("Equilateral Triangle!", font_size=28, color=GREEN)
        result.to_edge(DOWN, buff=0.5)
        self.play(Write(result))
        
        self.wait(2)
```

```python
            Text("Sum of\nAngles", font_size=18, color=YELLOW),
            Text("Each\nAngle", font_size=18, color=YELLOW),
            Text("Exterior\nAngle", font_size=18, color=YELLOW),
        )
        
        # Position headers
        for i, h in enumerate(headers):
            h.move_to(LEFT * 5 + RIGHT * i * 2.5 + UP * 1.5)
        
        self.play(Write(headers))

        data = [
            (3, "Triangle", "180°", "60°", "120°"),

        ).arrange(RIGHT, buff=0.5)
        formulas.to_edge(DOWN, buff=0.4)
        
        self.play(Write(formulas))
        
        self.wait(2)
```
