## UPPER AND LOWER BOUNDS CONCEPT
```python
# Introduction to bounds
title = Text("Upper and Lower Bounds", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Scenario
scenario = Text("A length is measured as 25 cm to the nearest cm", font_size=28, color=WHITE)
scenario.shift(UP * 1.5)

# Number line
number_line = NumberLine(
    x_range=[24, 26, 0.5],
    length=10,
    include_numbers=True,
    numbers_with_elongated_ticks=[24.5, 25, 25.5],
    font_size=24
).shift(DOWN * 0.2)

# Shaded region
bounds_region = Rectangle(
    width=5, height=0.5, 
    color=GREEN, fill_opacity=0.3
)
bounds_region.move_to(number_line.n2p(25))

# Lower bound
lb = MathTex("24.5", font_size=32, color=RED)
lb_label = Text("Lower Bound (LB)", font_size=20, color=RED)
lb.next_to(number_line.n2p(24.5), DOWN * 2)
lb_label.next_to(lb, DOWN, buff=0.1)

# Upper bound
ub = MathTex("25.5", font_size=32, color=BLUE)
ub_label = Text("Upper Bound (UB)", font_size=20, color=BLUE)
ub.next_to(number_line.n2p(25.5), DOWN * 2)
ub_label.next_to(ub, DOWN, buff=0.1)

# Formula
formula = MathTex("\\text{LB} \\leq \\text{actual value} < \\text{UB}", font_size=32, color=ORANGE)
formula.to_edge(DOWN, buff=0.8)

self.play(Write(title), run_time=0.8)
self.play(Write(scenario), run_time=1.0)
self.play(Create(number_line), run_time=1.0)
self.play(FadeIn(bounds_region), run_time=0.8)
self.play(Write(lb), Write(lb_label), run_time=1.0)
self.play(Write(ub), Write(ub_label), run_time=1.0)
self.play(Write(formula), run_time=1.0)
```

## BOUNDS IN ADDITION
```python
# Maximum and minimum for A + B
title = Text("Bounds: Addition", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Bounds
a_bounds = MathTex("A: 4.5 \\leq A < 5.5", font_size=28, color=BLUE).shift(UP * 0.7 + LEFT * 3)
b_bounds = MathTex("B: 2.5 \\leq B < 3.5", font_size=28, color=RED).shift(UP * 0.7 + RIGHT * 3)

# Maximum sum
max_calc = VGroup(
    Text("Maximum A + B", font_size=24, color=GREEN),
    MathTex("= 5.5 + 3.5", font_size=32),
    MathTex("= 9.0", font_size=36, color=GREEN)
).arrange(DOWN, buff=0.2)

# Minimum sum

    Text("Minimum A + B", font_size=24, color=ORANGE),
    MathTex("= 4.5 + 2.5", font_size=32),
    MathTex("= 7.0", font_size=36, color=ORANGE)
).arrange(DOWN, buff=0.2)

# Rule
rule = VGroup(
    Rectangle(width=9, height=0.8, color=YELLOW, fill_opacity=0.2),
    Text("MAX: UB + UB    MIN: LB + LB", font_size=24, color=YELLOW)

rule[1].move_to(rule[0])
rule.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
```

## BOUNDS SUMMARY
```python
# All rules on one screen
title = Text("Bounds Summary", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Table
rules = VGroup(
    VGroup(Text("Operation", font_size=24), Text("Maximum", font_size=24, color=GREEN), Text("Minimum", font_size=24, color=ORANGE)).arrange(RIGHT, buff=2),
    VGroup(MathTex("A + B", font_size=28), MathTex("UB + UB", font_size=28, color=GREEN), MathTex("LB + LB", font_size=28, color=ORANGE)).arrange(RIGHT, buff=2),
    VGroup(MathTex("A - B", font_size=28), MathTex("UB - LB", font_size=28, color=GREEN), MathTex("LB - UB", font_size=28, color=ORANGE)).arrange(RIGHT, buff=2),
    VGroup(MathTex("A \\times B", font_size=28), MathTex("UB \\times UB", font_size=28, color=GREEN), MathTex("LB \\times LB", font_size=28, color=ORANGE)).arrange(RIGHT, buff=2),
    VGroup(MathTex("A \\div B", font_size=28), MathTex("\\frac{UB}{LB}", font_size=28, color=GREEN), MathTex("\\frac{LB}{UB}", font_size=28, color=ORANGE)).arrange(RIGHT, buff=2),
    VGroup(MathTex("A^2", font_size=28), MathTex("UB^2", font_size=28, color=GREEN), MathTex("LB^2", font_size=28, color=ORANGE)).arrange(RIGHT, buff=2)
).arrange(DOWN, buff=0.35)
rules.move_to(ORIGIN)

# Tip
tip = VGroup(
    Rectangle(width=10, height=0.6, color=BLUE, fill_opacity=0.2),
    Text("Remember: Upper bound is NOT included (use < not ≤)", font_size=20, color=BLUE)
)
tip[1].move_to(tip[0])
tip.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
for rule in rules:
    self.play(Write(rule), run_time=0.5)
self.play(Create(tip), run_time=1.0)
```
