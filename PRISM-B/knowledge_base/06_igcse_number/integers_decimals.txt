## PLACE VALUE VISUALIZATION
```python
# Show place value columns
title = Text("Place Value", font_size=40, color=YELLOW).to_edge(UP, buff=0.5)

# Number to display: 3456.78
number_text = MathTex("3456.78", font_size=72, color=WHITE).move_to(UP * 1)

# Place value columns
columns = VGroup()
values = ["Thousands", "Hundreds", "Tens", "Ones", ".", "Tenths", "Hundredths"]
digits = ["3", "4", "5", "6", ".", "7", "8"]
colors = [RED, ORANGE, YELLOW, GREEN, WHITE, BLUE, PURPLE]

for i, (val, dig, col) in enumerate(zip(values, digits, colors)):
    if dig == ".":
        continue
    box = Rectangle(width=1.2, height=0.8, color=col, fill_opacity=0.3)
    digit = MathTex(dig, font_size=36, color=col)
    label = Text(val, font_size=14, color=col)
    digit.move_to(box)
    label.next_to(box, DOWN, buff=0.1)
    group = VGroup(box, digit, label)
    columns.add(group)

columns.arrange(RIGHT, buff=0.2)
columns.move_to(DOWN * 1)

self.play(Write(title), run_time=0.8)
self.play(Write(number_text), run_time=1.0)
for col in columns:
    self.play(Create(col), run_time=0.5)
```

## ROUNDING TO DECIMAL PLACES
```python
# Rounding 3.847 to different decimal places
title = Text("Rounding Decimal Places", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

number = MathTex("3.847", font_size=56).move_to(UP * 1.5)

# Underline the deciding digit for each case
steps = VGroup(
    VGroup(
        Text("To 2 d.p.: Look at 3rd decimal", font_size=24),
        MathTex(r"3.84\underline{7} \rightarrow 3.85", font_size=36, color=GREEN)
    ).arrange(DOWN, buff=0.2),
    VGroup(
        Text("To 1 d.p.: Look at 2nd decimal", font_size=24),
        MathTex(r"3.8\underline{4}7 \rightarrow 3.8", font_size=36, color=BLUE)
    ).arrange(DOWN, buff=0.2),
    VGroup(
        Text("To whole number: Look at 1st decimal", font_size=24),
        MathTex(r"3.\underline{8}47 \rightarrow 4", font_size=36, color=RED)
    ).arrange(DOWN, buff=0.2)
).arrange(DOWN, buff=0.5, aligned_edge=LEFT)
steps.move_to(DOWN * 0.5)

# Rule box
rule = VGroup(
    Rectangle(width=6, height=1.2, color=YELLOW, fill_opacity=0.1),
    Text("If digit ≥ 5, round UP\nIf digit < 5, round DOWN", font_size=22, color=YELLOW)
)
rule[1].move_to(rule[0])
rule.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
self.play(Write(number), run_time=1.0)
for step in steps:
    self.play(Write(step), run_time=1.5)
self.play(Create(rule), run_time=1.0)
```

## SIGNIFICANT FIGURES
```python
# Counting significant figures
title = Text("Significant Figures", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

examples = VGroup(
    VGroup(
        MathTex("0.00456", font_size=40),
        MathTex(r"\rightarrow 3 \text{ s.f.}", font_size=32, color=GREEN),

    VGroup(
        MathTex("3.040", font_size=40),
        MathTex(r"\rightarrow 4 \text{ s.f.}", font_size=32, color=GREEN),

    VGroup(
        MathTex("12000", font_size=40),

self.play(Write(title), run_time=0.8)

    self.play(Write(ex), run_time=1.2)
```

## NUMBER LINE OPERATIONS
```python
# Adding integers on number line
title = Text("Adding Integers on Number Line", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Number line from -5 to 10
line = NumberLine(
    x_range=[-5, 10, 1],
    length=12,
    include_numbers=True,
    include_tip=True,
    font_size=24
).shift(DOWN * 0.5)

# Show 3 + 5 = 8
start_dot = Dot(line.n2p(3), color=BLUE, radius=0.15)
start_label = MathTex("3", font_size=28, color=BLUE).next_to(start_dot, UP, buff=0.3)

# Arrow showing +5
arrow = Arrow(line.n2p(3), line.n2p(8), color=GREEN, buff=0, stroke_width=4)
arrow.shift(UP * 0.5)
plus_label = MathTex("+5", font_size=28, color=GREEN).next_to(arrow, UP, buff=0.1)

end_dot = Dot(line.n2p(8), color=RED, radius=0.15)
end_label = MathTex("8", font_size=28, color=RED).next_to(end_dot, DOWN, buff=0.3)

equation = MathTex("3 + 5 = 8", font_size=48, color=WHITE).to_edge(DOWN, buff=0.8)

self.play(Write(title), run_time=0.8)
self.play(Create(line), run_time=1.5)
self.play(Create(start_dot), Write(start_label), run_time=0.8)
self.play(Create(arrow), Write(plus_label), run_time=1.0)
self.play(Create(end_dot), Write(end_label), run_time=0.8)
self.play(Write(equation), run_time=1.0)
```

## NEGATIVE NUMBERS SUBTRACTION
```python
# Subtracting negative: 5 - (-3) = 5 + 3 = 8
title = Text("Subtracting Negatives", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Number line
line = NumberLine(
    x_range=[-2, 10, 1],
    length=10,
    include_numbers=True,
    include_tip=True,
    font_size=24
).shift(DOWN * 0.5)

# Step by step
step1 = MathTex(r"5 - (-3)", font_size=48).shift(UP * 2)
step2 = MathTex(r"= 5 + 3", font_size=48, color=GREEN).shift(UP * 2)
step3 = MathTex(r"= 8", font_size=48, color=BLUE).shift(UP * 2)

# Visual on number line
start = Dot(line.n2p(5), color=RED, radius=0.15)
arrow = Arrow(line.n2p(5), line.n2p(8), color=GREEN, buff=0).shift(UP * 0.4)
end = Dot(line.n2p(8), color=BLUE, radius=0.15)

rule = Text("Minus a negative = Plus a positive", font_size=24, color=YELLOW)
rule.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
self.play(Create(line), run_time=1.0)
self.play(Write(step1), run_time=1.0)
self.play(Create(start), run_time=0.5)
self.play(Transform(step1, step2), run_time=1.0)
self.play(Create(arrow), run_time=1.0)
self.play(Transform(step1, step3), Create(end), run_time=1.0)
self.play(Write(rule), run_time=1.0)
```

## ORDER OF OPERATIONS (BODMAS/BIDMAS)
```python
# BODMAS visualization
title = Text("Order of Operations: BODMAS", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# BODMAS breakdown
bodmas = VGroup(
    VGroup(Text("B", font_size=48, color=RED), Text("rackets", font_size=28)).arrange(RIGHT, buff=0.1),
    VGroup(Text("O", font_size=48, color=ORANGE), Text("rders (powers)", font_size=28)).arrange(RIGHT, buff=0.1),
    VGroup(Text("D", font_size=48, color=YELLOW), Text("ivision", font_size=28)).arrange(RIGHT, buff=0.1),
    VGroup(Text("M", font_size=48, color=GREEN), Text("ultiplication", font_size=28)).arrange(RIGHT, buff=0.1),
    VGroup(Text("A", font_size=48, color=BLUE), Text("ddition", font_size=28)).arrange(RIGHT, buff=0.1),
    VGroup(Text("S", font_size=48, color=PURPLE), Text("ubtraction", font_size=28)).arrange(RIGHT, buff=0.1),
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
bodmas.shift(LEFT * 3)

# Example: 3 + 4 × 2
example = VGroup(
    MathTex("3 + 4 \\times 2", font_size=40),
    MathTex("= 3 + 8", font_size=40, color=GREEN),
    MathTex("= 11", font_size=40, color=BLUE)
).arrange(DOWN, buff=0.3)
example.shift(RIGHT * 3 + UP * 1)

# Wrong vs Right
wrong = VGroup(
    Text("WRONG:", font_size=24, color=RED),
    MathTex("(3+4) \\times 2 = 14", font_size=32, color=RED)
).arrange(RIGHT, buff=0.3)
wrong.shift(RIGHT * 3 + DOWN * 1.5)

self.play(Write(title), run_time=0.8)
for item in bodmas:
    self.play(Write(item), run_time=0.5)
for step in example:
    self.play(Write(step), run_time=1.0)
self.play(Write(wrong), run_time=1.0)
```

## DECIMAL MULTIPLICATION
```python
# 2.4 × 0.3 = 0.72
title = Text("Multiplying Decimals", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Method: count decimal places
steps = VGroup(
    MathTex("2.4 \\times 0.3", font_size=48),
    Text("Step 1: Ignore decimals, multiply", font_size=24, color=GRAY),
    MathTex("24 \\times 3 = 72", font_size=40, color=BLUE),
    Text("Step 2: Count total decimal places", font_size=24, color=GRAY),
    MathTex("2.\\underline{4} \\text{ (1 d.p.)} + 0.\\underline{3} \\text{ (1 d.p.)} = \\text{2 d.p.}", font_size=32, color=GREEN),
    Text("Step 3: Place decimal in answer", font_size=24, color=GRAY),
    MathTex("72 \\rightarrow 0.72", font_size=48, color=RED)
).arrange(DOWN, buff=0.4)
steps.move_to(ORIGIN)

self.play(Write(title), run_time=0.8)
for step in steps:
    self.play(Write(step), run_time=1.0)
```
