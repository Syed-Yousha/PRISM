## SIMPLIFYING RATIOS
```python
# Simplify 24:36 to 2:3
title = Text("Simplifying Ratios", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Original ratio
original = MathTex("24 : 36", font_size=56).shift(UP * 1.5)

# Step by step
steps = VGroup(
    MathTex("\\div 2 \\rightarrow 12 : 18", font_size=40),
    MathTex("\\div 2 \\rightarrow 6 : 9", font_size=40),
    MathTex("\\div 3 \\rightarrow 2 : 3", font_size=40, color=GREEN)
).arrange(DOWN, buff=0.4)
steps.move_to(DOWN * 0.5)

# Or divide by HCF
alternative = VGroup(
    Text("Or find HCF(24, 36) = 12", font_size=24, color=GRAY),
    MathTex("24 \\div 12 : 36 \\div 12 = 2 : 3", font_size=36, color=GREEN)
).arrange(DOWN, buff=0.2)
alternative.to_edge(DOWN, buff=0.8)

self.play(Write(title), run_time=0.8)
self.play(Write(original), run_time=1.0)
for step in steps:
    self.play(Write(step), run_time=1.0)
self.play(Write(alternative), run_time=1.5)
```

## RATIO BAR MODEL
```python
# Sharing $60 in ratio 2:3
title = Text("Sharing in a Ratio", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

problem = Text("Share $60 in the ratio 2:3", font_size=28, color=WHITE).shift(UP * 2)

# Bar model - 5 equal parts
bars = VGroup()
# Person A gets 2 parts (blue)
for i in range(2):
    bar = Rectangle(width=1.5, height=0.8, fill_opacity=0.7, fill_color=BLUE, color=WHITE)
    bars.add(bar)
# Person B gets 3 parts (red)
for i in range(3):
    bar = Rectangle(width=1.5, height=0.8, fill_opacity=0.7, fill_color=RED, color=WHITE)
    bars.add(bar)

bars.arrange(RIGHT, buff=0.1)
bars.move_to(UP * 0.5)

# Labels
label_a = Text("Person A", font_size=24, color=BLUE).next_to(bars[:2], DOWN, buff=0.3)
label_b = Text("Person B", font_size=24, color=RED).next_to(bars[2:], DOWN, buff=0.3)

# Calculations
calc = VGroup(
    MathTex("\\text{Total parts} = 2 + 3 = 5", font_size=32),
    MathTex("\\text{1 part} = \\$60 \\div 5 = \\$12", font_size=32, color=GREEN),
    MathTex("\\text{Person A} = 2 \\times \\$12 = \\$24", font_size=32, color=BLUE),
    MathTex("\\text{Person B} = 3 \\times \\$12 = \\$36", font_size=32, color=RED)
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
calc.shift(DOWN * 1.8)

self.play(Write(title), run_time=0.8)
self.play(Write(problem), run_time=1.0)
self.play(*[Create(bar) for bar in bars], run_time=1.5)
self.play(Write(label_a), Write(label_b), run_time=0.8)
for c in calc:
    self.play(Write(c), run_time=1.0)
```

## DIRECT PROPORTION GRAPH
```python
# y is directly proportional to x
title = Text("Direct Proportion", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Coordinate system
axes = Axes(
    x_range=[0, 6, 1],
    y_range=[0, 12, 2],
    x_length=6,
    y_length=5,
    axis_config={"include_tip": True, "include_numbers": True}
).shift(LEFT * 2)

x_label = axes.get_x_axis_label("x", direction=RIGHT)
y_label = axes.get_y_axis_label("y", direction=UP)

# Line y = 2x (through origin)
line = axes.plot(lambda x: 2*x, color=BLUE, x_range=[0, 5.5])

# Key feature: passes through origin
origin_dot = Dot(axes.c2p(0, 0), color=RED, radius=0.15)
origin_label = Text("Through origin", font_size=20, color=RED).next_to(origin_dot, DR, buff=0.2)

# Formula
formula = VGroup(
    MathTex("y \\propto x", font_size=36),
    MathTex("y = kx", font_size=36, color=GREEN),
    Text("k = constant of proportionality", font_size=20, color=GRAY)
).arrange(DOWN, buff=0.3)
formula.shift(RIGHT * 3.5)

# Example values
example = VGroup(
    MathTex("\\text{If } y = 2x:", font_size=28),
    MathTex("x=1 \\rightarrow y=2", font_size=24),
    MathTex("x=2 \\rightarrow y=4", font_size=24),
    MathTex("x=3 \\rightarrow y=6", font_size=24)
).arrange(DOWN, buff=0.2)
example.shift(RIGHT * 3.5 + DOWN * 1.5)

self.play(Write(title), run_time=0.8)
self.play(Create(axes), Write(x_label), Write(y_label), run_time=1.5)
self.play(Create(line), run_time=1.0)
self.play(Create(origin_dot), Write(origin_label), run_time=0.8)
self.play(Write(formula), run_time=1.5)
self.play(Write(example), run_time=1.5)
```

## INVERSE PROPORTION GRAPH
```python
# y is inversely proportional to x
title = Text("Inverse Proportion", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Coordinate system
axes = Axes(
    x_range=[0, 6, 1],
    y_range=[0, 12, 2],
    x_length=6,
    y_length=5,
    axis_config={"include_tip": True, "include_numbers": True}
).shift(LEFT * 2)

x_label = axes.get_x_axis_label("x", direction=RIGHT)
y_label = axes.get_y_axis_label("y", direction=UP)

# Curve y = 12/x (hyperbola)
curve = axes.plot(lambda x: 12/x, color=RED, x_range=[1, 5.5])

# Formula
formula = VGroup(
    MathTex("y \\propto \\frac{1}{x}", font_size=36),
    MathTex("y = \\frac{k}{x}", font_size=36, color=RED),
    MathTex("xy = k \\text{ (constant)}", font_size=28, color=GREEN)
).arrange(DOWN, buff=0.3)
formula.shift(RIGHT * 3.5 + UP * 0.5)

# Key property
property_box = VGroup(
    Rectangle(width=4, height=1.2, color=YELLOW, fill_opacity=0.1),
    Text("As x increases,\ny decreases", font_size=22, color=YELLOW)
)
property_box[1].move_to(property_box[0])
property_box.shift(RIGHT * 3.5 + DOWN * 2)

# Points showing xy = constant
points = VGroup(
    Dot(axes.c2p(1, 12), color=WHITE, radius=0.1),
    Dot(axes.c2p(2, 6), color=WHITE, radius=0.1),
    Dot(axes.c2p(3, 4), color=WHITE, radius=0.1),
    Dot(axes.c2p(4, 3), color=WHITE, radius=0.1)
)

self.play(Write(title), run_time=0.8)
self.play(Create(axes), Write(x_label), Write(y_label), run_time=1.5)
self.play(Create(curve), run_time=1.5)
self.play(Create(points), run_time=1.0)
self.play(Write(formula), run_time=1.5)
self.play(Create(property_box), run_time=1.0)
```

## RATIO IN SIMILAR SHAPES
```python
# Similar triangles with ratio 1:2
title = Text("Ratio in Similar Shapes", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Small triangle (scale 1)
tri1 = Polygon(
    ORIGIN, RIGHT * 2, RIGHT * 2 + UP * 1.5,
    fill_opacity=0.3, fill_color=BLUE, color=BLUE
).shift(LEFT * 4)

# Large triangle (scale 2)
tri2 = Polygon(
    ORIGIN, RIGHT * 4, RIGHT * 4 + UP * 3,
    fill_opacity=0.3, fill_color=RED, color=RED
).shift(RIGHT * 1)

# Labels
small_label = Text("Small", font_size=24, color=BLUE).next_to(tri1, DOWN)
large_label = Text("Large", font_size=24, color=RED).next_to(tri2, DOWN)

# Dimensions
dim1 = MathTex("2", font_size=24, color=BLUE).next_to(tri1, DOWN, buff=0.5)
dim2 = MathTex("4", font_size=24, color=RED).next_to(tri2, DOWN, buff=0.5)

# Ratios
ratio_text = VGroup(
    Text("Length ratio = 1 : 2", font_size=28, color=WHITE),
    Text("Area ratio = 1² : 2² = 1 : 4", font_size=28, color=GREEN),
    Text("Volume ratio = 1³ : 2³ = 1 : 8", font_size=28, color=ORANGE)
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
ratio_text.to_edge(DOWN, buff=0.8)

self.play(Write(title), run_time=0.8)
self.play(Create(tri1), Write(small_label), run_time=1.0)
self.play(Create(tri2), Write(large_label), run_time=1.0)
for ratio in ratio_text:
    self.play(Write(ratio), run_time=1.0)
```

## THREE-PART RATIO
```python
# Sharing in ratio 2:3:5
title = Text("Three-Part Ratio", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

problem = Text("Share $500 in the ratio 2:3:5", font_size=28).shift(UP * 2)

# Bar model - 10 total parts
bars = VGroup()
colors = [BLUE, BLUE, GREEN, GREEN, GREEN, RED, RED, RED, RED, RED]
for c in colors:
    bar = Rectangle(width=0.8, height=0.6, fill_opacity=0.7, fill_color=c, color=WHITE)
    bars.add(bar)

bars.arrange(RIGHT, buff=0.05)
bars.shift(UP * 0.8)

# Labels
brace_a = Brace(bars[:2], DOWN, color=BLUE)
brace_b = Brace(bars[2:5], DOWN, color=GREEN)
brace_c = Brace(bars[5:], DOWN, color=RED)
label_a = Text("A: 2 parts", font_size=20, color=BLUE).next_to(brace_a, DOWN)
label_b = Text("B: 3 parts", font_size=20, color=GREEN).next_to(brace_b, DOWN)
label_c = Text("C: 5 parts", font_size=20, color=RED).next_to(brace_c, DOWN)

# Calculations
calc = VGroup(
    MathTex("\\text{Total} = 2+3+5 = 10 \\text{ parts}", font_size=28),
    MathTex("1 \\text{ part} = \\$500 \\div 10 = \\$50", font_size=28, color=YELLOW),
    MathTex("A = 2 \\times \\$50 = \\$100", font_size=28, color=BLUE),
    MathTex("B = 3 \\times \\$50 = \\$150", font_size=28, color=GREEN),
    MathTex("C = 5 \\times \\$50 = \\$250", font_size=28, color=RED)
).arrange(DOWN, buff=0.25, aligned_edge=LEFT)
calc.shift(DOWN * 1.5)

self.play(Write(title), run_time=0.8)
self.play(Write(problem), run_time=1.0)
self.play(*[Create(bar) for bar in bars], run_time=1.5)
self.play(Create(brace_a), Create(brace_b), Create(brace_c), run_time=0.8)
self.play(Write(label_a), Write(label_b), Write(label_c), run_time=0.8)
for c in calc:
    self.play(Write(c), run_time=0.8)
```
