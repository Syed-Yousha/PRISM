## PHILOSOPHY
PRISM videos are EDUCATIONAL. Every animation serves a PURPOSE.
- No random shapes floating around
- Every element explains something
- Smooth, deliberate transitions
- Clean, uncluttered frames

## COLOR PALETTE (MANDATORY)
```python
# PRIMARY COLORS - Use these exactly
BACKGROUND = "#000000"      # Pure black always
TITLE_COLOR = YELLOW        # Section titles, emphasis
PRIMARY = BLUE              # Main shapes, key elements
SECONDARY = TEAL            # Supporting elements, labels
TEXT_COLOR = WHITE          # Body text, formulas
SUCCESS = GREEN             # Correct answers, positive highlights
WARNING = RED               # Errors, attention points

# USAGE EXAMPLES
title = Text("Quadratic Formula", color=YELLOW, font_size=48)
formula = MathTex(r"x^2 + y^2 = r^2", color=WHITE)
shape = Circle(color=BLUE, fill_opacity=0.3)
answer = MathTex(r"x = 5", color=GREEN)
```

## TYPOGRAPHY RULES
```python
# FONT SIZES - Consistent hierarchy
TITLE_SIZE = 48       # Main section titles
HEADING_SIZE = 40     # Sub-sections
BODY_SIZE = 32        # Regular text
SMALL_SIZE = 24       # Labels, annotations
TINY_SIZE = 18        # Footnotes, axis labels

# MATH - ALWAYS use MathTex with raw strings
MathTex(r"E = mc^2", font_size=48)                    # Simple
MathTex(r"\frac{-b \pm \sqrt{b^2-4ac}}{2a}")          # Complex
MathTex(r"\int_0^1 x^2 \, dx = \frac{1}{3}")          # Calculus

# TEXT - Use Text() for plain text
Text("Key Points:", font_size=40, color=BLUE)
Text("• First point", font_size=28, color=WHITE)

# NEVER DO THIS:
# Tex(r"\bullet")        # CRASHES! Use Text("•") instead
# MathTex(r"$x^2$")      # WRONG! No $ inside MathTex
# Text("x²")             # Use MathTex for math symbols
```

## SCREEN LAYOUT
```
┌──────────────────────────────────────────────────────────────┐
│                    TITLE (YELLOW, UP edge)                   │
│                                                              │
│                                                              │
│                    MAIN CONTENT AREA                         │
│                    (CENTER or RIGHT*2)                       │
│                                                              │
│                    Formulas, Graphs, Shapes                  │
│                                                              │
│                                                              │
└──────────────────────────────────────────────────────────────┘

# POSITIONING CONSTANTS
TITLE_POS = UP * 3.5           # to_edge(UP)
CENTER_POS = ORIGIN            # Center of screen
RIGHT_AREA = RIGHT * 2.5       # Main content area
LEFT_LABEL = LEFT * 5          # Small labels only
```

## ANIMATION TIMING
```python
# STANDARD DURATIONS (feel natural, not rushed)
WRITE_SPEED = 1.5              # Text appearing
CREATE_SPEED = 1.0             # Shapes appearing
TRANSFORM_SPEED = 1.5          # Equation morphing
INDICATE_SPEED = 0.5           # Quick highlight
FADEOUT_SPEED = 0.5            # Clearing screen
WAIT_SHORT = 0.5               # Brief pause
WAIT_MEDIUM = 1.0              # Standard pause
WAIT_LONG = 2.0                # Emphasis pause

# EXAMPLE FLOW
self.play(Write(title), run_time=1.5)
self.wait(0.5)
self.play(Write(formula), run_time=2.0)
self.play(Indicate(formula, color=YELLOW), run_time=0.5)
self.wait(1.0)
```

## SECTION TRANSITIONS
```python
# ALWAYS clear before new section (except first)
def clear_screen(self):
    self.play(*[FadeOut(mob) for mob in self.mobjects], run_time=0.5)

# PATTERN FOR EACH SECTION:
# 1. Clear previous (if not first)
# 2. Show title
# 3. Build content step by step
# 4. Wait for narration
# 5. Optional highlight key point
```

## 3D SCENE RULES
```python
# For ThreeDScene, ALWAYS add text to fixed frame
class MyScene(ThreeDScene):
    def construct(self):
        # Text must be fixed to frame in 3D
        title = Text("3D Visualization", color=YELLOW)
        self.add_fixed_in_frame_mobjects(title)
        title.to_edge(UP)
        self.play(Write(title))
        
        # 3D objects can be added normally
        sphere = Sphere(radius=1, color=BLUE)
        self.play(Create(sphere))
```
