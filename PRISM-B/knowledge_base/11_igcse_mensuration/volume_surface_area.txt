## 1. Volume and Surface Area of Cuboids

```python
from manim import *
import numpy as np

class CuboidVolumeSA(ThreeDScene):
    def construct(self):
        self.set_camera_orientation(phi=70 * DEGREES, theta=-45 * DEGREES)
        
        title = Text("Cuboid: Volume & Surface Area", font_size=32, color=BLUE)
        title.to_edge(UP)
        title.fix_in_frame()
        self.add(title)
        
        # Cuboid
        cuboid = Prism(dimensions=[3, 2, 1.5], fill_color=YELLOW, fill_opacity=0.5)
        
        self.play(Create(cuboid))
        self.wait(0.5)
        
        # Labels
        length_label = MathTex("l = 6", font_size=24, color=WHITE)
        width_label = MathTex("w = 4", font_size=24, color=WHITE)
        height_label = MathTex("h = 3", font_size=24, color=WHITE)
        
        length_label.next_to(cuboid, DOWN + RIGHT)
        width_label.next_to(cuboid, DOWN + LEFT)
        height_label.next_to(cuboid, UP)
        
        for label in [length_label, width_label, height_label]:
            label.fix_in_frame()
        
        self.add(length_label, width_label, height_label)
        
        # Formulas (fixed to frame)
        formulas = VGroup(
            Text("Volume = l × w × h", font_size=22),
            MathTex(r"= 6 \times 4 \times 3 = 72 \text{ cm}^3", font_size=22, color=GREEN),
            Text("Surface Area = 2(lw + lh + wh)", font_size=20),
            MathTex(r"= 2(24 + 18 + 12) = 108 \text{ cm}^2", font_size=22, color=GREEN),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.2)
        formulas.to_edge(RIGHT)
        formulas.shift(DOWN * 0.5)
        formulas.fix_in_frame()
        
        self.add(formulas)
        
        # Rotate for better view
        self.begin_ambient_camera_rotation(rate=0.2)
        self.wait(3)
        self.stop_ambient_camera_rotation()
```

## 5. Volume and Surface Area of Pyramids

```python
from manim import *
import numpy as np

class PyramidVolumeSA(Scene):
    def construct(self):
        title = Text("Pyramid: Volume & Surface Area", font_size=32, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Square-based pyramid
        # Base
        base = Square(side_length=3, color=YELLOW, fill_opacity=0.2)
        base.shift(LEFT * 3 + DOWN * 1.5)
        base.rotate(PI/6)  # Slight rotation for 3D effect
        
        # Apex
        apex = LEFT * 3 + UP * 1.5
        
        # Edges
        corners = base.get_vertices()
        edges = VGroup()
        for corner in corners:
            edge = Line(corner, apex, color=YELLOW)
            edges.add(edge)
        
        self.play(Create(base), Create(edges))
        
        # Labels
        base_label = MathTex("\\text{base} = 6 \\times 6", font_size=20, color=WHITE)
        base_label.next_to(base, DOWN, buff=0.3)
        
        h_label = MathTex("h = 4", font_size=20, color=GREEN)
        h_label.shift(LEFT * 2.3)
        
        # Height line
        h_line = DashedLine(apex, LEFT * 3 + DOWN * 1.5, color=GREEN)
        self.play(Create(h_line), Write(h_label), Write(base_label))
        
        # Formulas
        formulas = VGroup(
            Text("Volume:", font_size=22, color=YELLOW, weight=BOLD),
            MathTex(r"V = \frac{1}{3} \times \text{base area} \times h", font_size=22),
            MathTex(r"= \frac{1}{3} \times 36 \times 4", font_size=22),
            MathTex(r"= 48 \text{ cm}^3", font_size=24, color=GREEN),
            Text(" ", font_size=10),
            Text("Surface Area:", font_size=22, color=YELLOW, weight=BOLD),
            Text("= Base + 4 triangular faces", font_size=18, color=GRAY),
            MathTex(r"SA = l^2 + 4 \times \frac{1}{2}bl_s", font_size=22),
            Text("(where l_s = slant height)", font_size=16, color=GRAY),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.12)
        formulas.shift(RIGHT * 2 + DOWN * 0.3)
        
        self.play(Write(formulas), run_time=2.5)
        
        # Key formula box
        key = MathTex(r"V_{\text{pyramid}} = \frac{1}{3} \times V_{\text{prism}}", font_size=24, color=ORANGE)
        key.to_edge(DOWN, buff=0.4)
        box = SurroundingRectangle(key, color=ORANGE)
        
        self.play(Write(key), Create(box))
        
        self.wait(2)
```

        )

        self.play(Create(small_cone), Create(small_base))

            LEFT * 1 + DOWN * 2, UP * 1.6, RIGHT * 1 + DOWN * 2,
            color=GREEN, fill_opacity=0.3
        )
        large_cone.shift(RIGHT * 0.5)
        large_base = Ellipse(width=2, height=0.6, color=GREEN)

        small_label.next_to(small_base, DOWN, buff=0.1)

        large_label.next_to(large_base, DOWN, buff=0.1)
        
        self.play(Write(small_label), Write(large_label))
        
        # Scale factor
        sf = MathTex(r"\text{Scale factor } k = \frac{6}{3} = 2", font_size=24, color=YELLOW)
        sf.shift(RIGHT * 3.5 + UP * 2)
        self.play(Write(sf))
        
        # Key relationships
        relations = VGroup(
            Text("If linear scale factor = k:", font_size=20, color=WHITE),
            MathTex(r"\text{Surface Area ratio} = k^2 = 4", font_size=22, color=ORANGE),
            MathTex(r"\text{Volume ratio} = k^3 = 8", font_size=22, color=GREEN),
