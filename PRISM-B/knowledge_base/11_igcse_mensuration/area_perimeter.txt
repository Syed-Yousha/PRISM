## 1. Area and Perimeter of Rectangles

```python
from manim import *
import numpy as np

class RectangleAreaPerimeter(Scene):
    def construct(self):
        title = Text("Rectangle: Area & Perimeter", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Rectangle
        rect = Rectangle(width=4, height=2.5, color=YELLOW, fill_opacity=0.3)
        rect.shift(LEFT * 2)
        
        self.play(Create(rect))
        
        # Labels
        length_label = MathTex("l = 8 \\text{ cm}", font_size=24, color=WHITE)
        length_label.next_to(rect, DOWN, buff=0.2)
        
        width_label = MathTex("w = 5 \\text{ cm}", font_size=24, color=WHITE)
        width_label.next_to(rect, RIGHT, buff=0.2)
        
        self.play(Write(length_label), Write(width_label))
        
        # Formulas and calculations
        formulas = VGroup(
            Text("Formulas:", font_size=24, color=YELLOW, weight=BOLD),
            MathTex(r"\text{Area} = l \times w", font_size=24),
            MathTex(r"= 8 \times 5 = 40 \text{ cm}^2", font_size=24, color=GREEN),
            MathTex(r"\text{Perimeter} = 2(l + w)", font_size=24),
            MathTex(r"= 2(8 + 5) = 26 \text{ cm}", font_size=24, color=GREEN),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.25)
        formulas.shift(RIGHT * 3)
        
        for line in formulas:
            self.play(Write(line), run_time=0.5)
        
        # Highlight area
        area_fill = rect.copy()
        area_fill.set_fill(GREEN, opacity=0.5)
        self.play(Transform(rect, area_fill))
        
        self.wait(2)
```
