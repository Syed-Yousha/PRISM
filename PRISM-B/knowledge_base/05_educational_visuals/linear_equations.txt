## BASIC COORDINATE PLANE
```python
# Create a standard coordinate plane
axes = Axes(
    x_range=[-5, 5, 1],
    y_range=[-4, 4, 1],
    x_length=10,
    y_length=8,
    axis_config={
        "include_tip": True,
        "include_numbers": True,
        "color": WHITE
    }
)

# Axis labels
x_label = axes.get_x_axis_label(r"x", direction=RIGHT)
y_label = axes.get_y_axis_label(r"y", direction=UP)

self.play(Create(axes), run_time=1.5)
self.play(Write(x_label), Write(y_label), run_time=0.8)
```

## PLOTTING A LINEAR EQUATION
```python
# Plot y = 2x + 1
axes = Axes(
    x_range=[-4, 4, 1],
    y_range=[-3, 5, 1],
    x_length=8,
    y_length=6,
    axis_config={"include_tip": True, "include_numbers": True}
)

# The line
line = axes.plot(lambda x: 2*x + 1, color=BLUE, x_range=[-2, 2])

# Y-intercept point (0, 1)
y_intercept = Dot(axes.c2p(0, 1), color=RED, radius=0.15)
y_int_label = MathTex("(0, 1)", font_size=28, color=RED).next_to(y_intercept, RIGHT, buff=0.2)

# Equation label
equation = MathTex(r"y = 2x + 1", font_size=40, color=BLUE).to_edge(UP, buff=0.5)

self.play(Create(axes), run_time=1.0)
self.play(Create(line), run_time=1.5)
self.play(Create(y_intercept), Write(y_int_label), run_time=1.0)
self.play(Write(equation), run_time=0.8)
```

## SLOPE VISUALIZATION (RISE OVER RUN)
```python
# Visualize slope = rise/run = 2/1
axes = Axes(
    x_range=[-4, 4, 1],
    y_range=[-2, 6, 1],
    x_length=8,
    y_length=6,
    axis_config={"include_tip": True}
)

# The line y = 2x + 1
line = axes.plot(lambda x: 2*x + 1, color=BLUE, x_range=[-1.5, 2.5])

# Two points on the line
point1 = Dot(axes.c2p(0, 1), color=WHITE, radius=0.12)
point2 = Dot(axes.c2p(1, 3), color=WHITE, radius=0.12)

# Rise (vertical line)
rise_line = Line(
    axes.c2p(1, 1), 
    axes.c2p(1, 3), 
    color=GREEN, 
    stroke_width=4
)
rise_label = MathTex(r"\text{rise} = 2", font_size=28, color=GREEN)
rise_label.next_to(rise_line, RIGHT, buff=0.2)

# Run (horizontal line)
run_line = Line(
    axes.c2p(0, 1), 
    axes.c2p(1, 1), 
    color=RED, 
    stroke_width=4
)
run_label = MathTex(r"\text{run} = 1", font_size=28, color=RED)
run_label.next_to(run_line, DOWN, buff=0.2)

# Slope formula
slope_formula = MathTex(
    r"m = \frac{\text{rise}}{\text{run}} = \frac{2}{1} = 2",
    font_size=36
).to_edge(DOWN, buff=0.5)

self.play(Create(axes), run_time=1.0)
self.play(Create(line), run_time=1.0)
self.play(Create(point1), Create(point2), run_time=0.8)
self.play(Create(run_line), Write(run_label), run_time=1.0)
self.play(Create(rise_line), Write(rise_label), run_time=1.0)
self.play(Write(slope_formula), run_time=1.0)
```

## SLOPE-INTERCEPT FORM (y = mx + b)
```python
# Show the meaning of m and b
title = Text("Slope-Intercept Form", font_size=40, color=YELLOW).to_edge(UP, buff=0.5)

# The formula with color coding
formula = MathTex(r"y = ", r"m", r"x + ", r"b", font_size=56)
formula[1].set_color(BLUE)   # m in blue
formula[3].set_color(RED)    # b in red
formula.move_to(UP * 1)

# Labels
m_label = Text("slope (steepness)", font_size=24, color=BLUE)
m_label.next_to(formula[1], DOWN, buff=0.8)
m_arrow = Arrow(m_label.get_top(), formula[1].get_bottom(), color=BLUE, buff=0.1)

b_label = Text("y-intercept", font_size=24, color=RED)
b_label.next_to(formula[3], DOWN, buff=0.8)
b_arrow = Arrow(b_label.get_top(), formula[3].get_bottom(), color=RED, buff=0.1)

self.play(Write(title), run_time=0.8)
self.play(Write(formula), run_time=1.5)
self.play(Create(m_arrow), Write(m_label), run_time=1.0)
self.play(Create(b_arrow), Write(b_label), run_time=1.0)
```

## COMPARING DIFFERENT SLOPES
```python
# Show lines with different slopes on same axes
axes = Axes(
    x_range=[-4, 4, 1],
    y_range=[-4, 4, 1],
    x_length=7,
    y_length=6,
    axis_config={"include_tip": True}
).shift(LEFT * 1.5)

# Lines with different slopes
line1 = axes.plot(lambda x: 0.5*x, color=GREEN, x_range=[-3, 3])  # m = 0.5
line2 = axes.plot(lambda x: x, color=BLUE, x_range=[-3, 3])        # m = 1
line3 = axes.plot(lambda x: 2*x, color=RED, x_range=[-2, 2])       # m = 2
line4 = axes.plot(lambda x: -x, color=PURPLE, x_range=[-3, 3])     # m = -1

# Legend
legend = VGroup(
    VGroup(Line(ORIGIN, RIGHT*0.5, color=GREEN), MathTex(r"m = 0.5", font_size=24, color=GREEN)).arrange(RIGHT, buff=0.2),
    VGroup(Line(ORIGIN, RIGHT*0.5, color=BLUE), MathTex(r"m = 1", font_size=24, color=BLUE)).arrange(RIGHT, buff=0.2),
    VGroup(Line(ORIGIN, RIGHT*0.5, color=RED), MathTex(r"m = 2", font_size=24, color=RED)).arrange(RIGHT, buff=0.2),
    VGroup(Line(ORIGIN, RIGHT*0.5, color=PURPLE), MathTex(r"m = -1", font_size=24, color=PURPLE)).arrange(RIGHT, buff=0.2),
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
legend.shift(RIGHT * 4)

self.play(Create(axes), run_time=1.0)
self.play(Create(line1), run_time=0.8)
self.play(Create(line2), run_time=0.8)
self.play(Create(line3), run_time=0.8)
self.play(Create(line4), run_time=0.8)
self.play(Write(legend), run_time=1.0)
```

## FINDING EQUATION FROM TWO POINTS
```python
# Given points (1, 2) and (3, 6), find the equation
title = Text("Find Equation from Two Points", font_size=32, color=YELLOW).to_edge(UP, buff=0.5)

# Show the points
axes = Axes(
    x_range=[-1, 5, 1],
    y_range=[-1, 8, 1],
    x_length=6,
    y_length=5
).shift(LEFT * 2 + DOWN * 0.5)

point1 = Dot(axes.c2p(1, 2), color=RED, radius=0.15)
point2 = Dot(axes.c2p(3, 6), color=RED, radius=0.15)
p1_label = MathTex("(1, 2)", font_size=24).next_to(point1, LEFT, buff=0.2)
p2_label = MathTex("(3, 6)", font_size=24).next_to(point2, RIGHT, buff=0.2)

# Line through points
line = axes.plot(lambda x: 2*x, color=BLUE, x_range=[0, 4])

# Step-by-step calculation on the right
step1 = MathTex(r"m = \frac{y_2 - y_1}{x_2 - x_1}", font_size=32).shift(RIGHT * 3 + UP * 1)
step2 = MathTex(r"m = \frac{6 - 2}{3 - 1} = \frac{4}{2} = 2", font_size=32).shift(RIGHT * 3 + UP * 0)
step3 = MathTex(r"y - y_1 = m(x - x_1)", font_size=32).shift(RIGHT * 3 + DOWN * 1)
step4 = MathTex(r"y - 2 = 2(x - 1)", font_size=32).shift(RIGHT * 3 + DOWN * 2)
step5 = MathTex(r"y = 2x", font_size=36, color=GREEN).shift(RIGHT * 3 + DOWN * 3)

self.play(Write(title), run_time=0.8)
self.play(Create(axes), run_time=1.0)
self.play(Create(point1), Create(point2), Write(p1_label), Write(p2_label), run_time=1.0)
self.play(Write(step1), run_time=1.0)
self.play(Write(step2), run_time=1.0)
self.play(Create(line), run_time=1.0)
self.play(Write(step3), run_time=1.0)
self.play(Write(step4), run_time=1.0)
self.play(Write(step5), run_time=1.0)
```

## PARALLEL AND PERPENDICULAR LINES
```python
# Show parallel lines (same slope) and perpendicular lines
axes = Axes(
    x_range=[-4, 4, 1],
    y_range=[-4, 4, 1],
    x_length=6,
    y_length=5
).shift(LEFT * 2)

# Parallel lines (slope = 1)
line1 = axes.plot(lambda x: x + 1, color=BLUE, x_range=[-3, 3])
line2 = axes.plot(lambda x: x - 1, color=BLUE, x_range=[-3, 3])
parallel_label = Text("Parallel: same slope", font_size=24, color=BLUE)
parallel_label.shift(RIGHT * 3 + UP * 1.5)

# Perpendicular line (slope = -1, negative reciprocal)
line3 = axes.plot(lambda x: -x, color=RED, x_range=[-3, 3])
perp_label = Text("Perpendicular: slopes", font_size=24, color=RED)
perp_formula = MathTex(r"m_1 \cdot m_2 = -1", font_size=28, color=RED)
perp_group = VGroup(perp_label, perp_formula).arrange(DOWN, buff=0.2)
perp_group.shift(RIGHT * 3 + DOWN * 1)

# Right angle marker at origin
right_angle = Square(side_length=0.3, color=WHITE, stroke_width=2)
right_angle.move_to(axes.c2p(0.15, -0.15))

self.play(Create(axes), run_time=1.0)
self.play(Create(line1), Create(line2), run_time=1.0)
self.play(Write(parallel_label), run_time=0.8)
self.play(Create(line3), run_time=1.0)
self.play(Create(right_angle), run_time=0.5)
self.play(Write(perp_group), run_time=1.0)
```
