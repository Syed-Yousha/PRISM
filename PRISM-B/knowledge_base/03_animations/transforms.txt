## TRANSFORMMATCHINGTEX - Equation Morphing
```python
# The signature 3b1b effect - equations morphing into each other
# Parts with matching LaTeX smoothly transform

# Basic transform between equations
eq1 = MathTex(r"a^2 + b^2 = c^2", font_size=48)
eq2 = MathTex(r"c = \sqrt{a^2 + b^2}", font_size=48)

self.play(Write(eq1), run_time=1.5)
self.wait(1.0)
self.play(TransformMatchingTex(eq1, eq2), run_time=2.0)
self.wait(1.0)

# Transform with color preservation
eq1 = MathTex(r"x^2", r"+", r"y^2", r"=", r"r^2", font_size=48)

eq1[2].set_color(GREEN)
eq1[4].set_color(YELLOW)

eq2 = MathTex(r"x^2", r"+", r"y^2", font_size=48)

eq2[2].set_color(GREEN)

self.play(Write(eq1), run_time=1.5)
self.wait(0.5)
self.play(TransformMatchingTex(eq1, eq2), run_time=1.5)
```

## REPLACEMENTTRANSFORM - Complete Replacement
```python
# When equations don't share parts, use ReplacementTransform
old_eq = MathTex(r"\frac{dy}{dx}", font_size=48)
new_eq = MathTex(r"2x", font_size=48)
new_eq.move_to(old_eq)  # Position at same location

self.play(Write(old_eq), run_time=1.5)
self.wait(0.5)
self.play(ReplacementTransform(old_eq, new_eq), run_time=1.5)

# Step-by-step derivation with ReplacementTransform
steps = [
    MathTex(r"y = x^2"),
    MathTex(r"\frac{dy}{dx} = 2x"),
    MathTex(r"y' = 2x"),
]

current = steps[0]
self.play(Write(current), run_time=1.5)

for next_step in steps[1:]:
    next_step.move_to(current)
    self.wait(0.5)
    self.play(ReplacementTransform(current, next_step), run_time=1.5)
    current = next_step
```

## COMPLETE EXAMPLE: Quadratic Derivation
```python
from manim import *

class QuadraticDerivation(Scene):
    def construct(self):
        # Title
        title = Text("Solving a Quadratic Equation", font_size=44, color=YELLOW)
        title.to_edge(UP)
        self.play(Write(title), run_time=1.0)
        
        # Starting equation
        eq1 = MathTex(r"x^2 + 5x + 6 = 0", font_size=44)
        eq1.move_to(UP * 0.5)
        self.play(Write(eq1), run_time=1.5)
        self.wait(1.0)
        
        # Factor form
        eq2 = MathTex(r"(x + 2)(x + 3) = 0", font_size=44)
        eq2.move_to(UP * 0.5)
        
        self.play(TransformMatchingTex(eq1, eq2), run_time=2.0)
        self.wait(1.0)
        
        # Split into two factors
        factor1 = MathTex(r"x + 2 = 0", font_size=40, color=BLUE)
        factor2 = MathTex(r"x + 3 = 0", font_size=40, color=TEAL)
        
        factor1.move_to(LEFT * 2.5 + DOWN * 1)
        factor2.move_to(RIGHT * 2.5 + DOWN * 1)
        
        self.play(Write(factor1), Write(factor2), run_time=1.5)
        self.wait(0.5)
        
        # Solutions
        sol1 = MathTex(r"x = -2", font_size=44, color=GREEN)
        sol2 = MathTex(r"x = -3", font_size=44, color=GREEN)
        
        sol1.move_to(LEFT * 2.5 + DOWN * 2.5)
        sol2.move_to(RIGHT * 2.5 + DOWN * 2.5)
        
        self.play(
            TransformMatchingTex(factor1.copy(), sol1),
            TransformMatchingTex(factor2.copy(), sol2),
            run_time=1.5
        )
        
        # Highlight answers
        box1 = SurroundingRectangle(sol1, color=GREEN, buff=0.2)
        box2 = SurroundingRectangle(sol2, color=GREEN, buff=0.2)
        
        self.play(Create(box1), Create(box2), run_time=0.5)
        self.wait(2.0)
```
