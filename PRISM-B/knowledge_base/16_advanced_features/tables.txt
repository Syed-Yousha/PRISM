## BASIC MATHTABLE
```python
# Simple table with math expressions
from manim import *

table = MathTable(
    [["+", "a", "b"],
     ["a", "a^2", "ab"],
     ["b", "ab", "b^2"]],
    include_outer_lines=True
)
table.scale(0.8)

# Highlight specific cells
table.add_highlighted_cell((2, 2), color=GREEN)
table.add_highlighted_cell((3, 3), color=BLUE)

self.play(Create(table))
self.wait()
```

## INTEGER TABLE (Multiplication Table)
```python
# Times table visualization
from manim import *

# Create a 5x5 multiplication table
table = IntegerTable(
    [[i * j for j in range(1, 6)] for i in range(1, 6)],
    row_labels=[MathTex(str(i)) for i in range(1, 6)],
    col_labels=[MathTex(str(j)) for j in range(1, 6)],
    include_outer_lines=True,
    h_buff=1.0,
    v_buff=0.8
)
table.scale(0.7)

# Add title
title = Text("Multiplication Table", font_size=36, color=YELLOW).to_edge(UP)

self.play(Write(title))
self.play(Create(table), run_time=2)
self.wait()

# Highlight diagonal (perfect squares)
for i in range(1, 6):
    cell = table.get_cell((i + 1, i + 1))  # +1 because of labels
    self.play(cell.animate.set_fill(YELLOW, opacity=0.5), run_time=0.3)
```

## TABLE WITH CUSTOM STYLING
```python
# Styled table for data presentation
from manim import *

# Create data table
data = [
    ["Student", "Math", "Science", "English"],
    ["Alice", "85", "92", "78"],
    ["Bob", "72", "88", "95"],
    ["Carol", "91", "76", "82"]
]

table = Table(
    data[1:],  # Data rows
    row_labels=[Text(row[0], font_size=24) for row in data[1:]],
    col_labels=[Text(col, font_size=24, color=YELLOW) for col in data[0][1:]],
    include_outer_lines=True,
    line_config={"stroke_width": 2, "color": WHITE},
    h_buff=0.8,
    v_buff=0.5
)
table.scale(0.7)

# Style header row
header_cells = table.get_row(1)
for cell in header_cells:
    cell.set_fill(BLUE, opacity=0.3)

title = Text("Student Grades", font_size=40, color=YELLOW).to_edge(UP)

self.play(Write(title))
self.play(Create(table), run_time=2)
self.wait()
```

## HIGHLIGHTING ROWS AND COLUMNS
```python
# Highlight specific rows/columns
from manim import *

table = MathTable(
    [["x", "f(x)", "f'(x)"],
     ["1", "3", "2"],
     ["2", "7", "4"],
     ["3", "13", "6"]],
    include_outer_lines=True
)
table.scale(0.8)

self.play(Create(table))
self.wait()

# Highlight column (derivative column)
col = table.get_columns()[2]
self.play(col.animate.set_color(GREEN))
self.wait()

# Highlight a specific row
row = table.get_rows()[2]
box = SurroundingRectangle(row, color=RED, buff=0.1)
self.play(Create(box))
self.wait()

# Get specific entry
entry = table.get_entries((3, 2))  # Row 3, Col 2
self.play(entry.animate.scale(1.5).set_color(YELLOW))
self.wait()
```

## ANIMATED TABLE FILLING
```python
# Fill table entries one by one
from manim import *

# Empty table structure
table = MathTable(
    [["x", "x^2", "2x"],
     ["1", "?", "?"],
     ["2", "?", "?"],
     ["3", "?", "?"]],
    include_outer_lines=True
)
table.scale(0.8)

self.play(Create(table))
self.wait()

# Replace entries with calculated values
replacements = [
    ((2, 2), "1"), ((2, 3), "2"),
    ((3, 2), "4"), ((3, 3), "4"),
    ((4, 2), "9"), ((4, 3), "6")
]

for pos, value in replacements:
    old_entry = table.get_entries(pos)
    new_entry = MathTex(value, color=GREEN).move_to(old_entry)
    self.play(Transform(old_entry, new_entry), run_time=0.5)
self.wait()
```

## TRUTH TABLE (Logic)
```python
# Boolean logic truth table
from manim import *

table = MathTable(
    [["P", "Q", r"P \land Q", r"P \lor Q", r"P \rightarrow Q"],
     ["T", "T", "T", "T", "T"],
     ["T", "F", "F", "T", "F"],
     ["F", "T", "F", "T", "T"],
     ["F", "F", "F", "F", "T"]],
    include_outer_lines=True,
    h_buff=1.2
)
table.scale(0.6)

title = Text("Truth Table", font_size=36, color=YELLOW).to_edge(UP)

# Color the True/False values
self.play(Write(title))
self.play(Create(table), run_time=2)

# Highlight all "T" entries in green, "F" in red
entries = table.get_entries()
for entry in entries:
    if hasattr(entry, 'tex_string'):
        if entry.tex_string == "T":
            entry.set_color(GREEN)
        elif entry.tex_string == "F":
            entry.set_color(RED)
self.wait()
```
