## 1. Matrix Basics - Order and Notation

```python
from manim import *
import numpy as np

class MatrixBasics(Scene):
    def construct(self):
        title = Text("Matrix Basics", font_size=40, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # What is a matrix
        intro = Text("A matrix is an array of numbers", font_size=24, color=WHITE)
        intro.next_to(title, DOWN, buff=0.3)
        self.play(Write(intro))
        
        # Example matrix
        matrix = Matrix([
            [2, 5, -1],
            [4, 0, 3]
        ], h_buff=1.2)
        matrix.shift(LEFT * 2)
        
        self.play(Create(matrix))
        
        # Order explanation
        order = VGroup(
            Text("Order: 2 × 3", font_size=26, color=YELLOW),
            Text("(rows × columns)", font_size=20, color=GRAY),
        ).arrange(DOWN, buff=0.1)
        order.next_to(matrix, RIGHT, buff=1)
        
        self.play(Write(order))
        
        # Row and column arrows
        row_arrow = Arrow(start=LEFT * 4.5 + UP * 0.3, end=LEFT * 0.5 + UP * 0.3, 
                         color=RED, buff=0)
        row_label = Text("2 rows", font_size=18, color=RED)
        row_label.next_to(row_arrow, UP, buff=0.1)
        
        col_arrow = Arrow(start=LEFT * 2.5 + UP * 1.3, end=LEFT * 2.5 + DOWN * 0.7,
                         color=GREEN, buff=0)
        col_label = Text("3 columns", font_size=18, color=GREEN)
        col_label.next_to(col_arrow, RIGHT, buff=0.1)
        
        self.play(Create(row_arrow), Write(row_label))
        self.play(Create(col_arrow), Write(col_label))
        
        # Element notation
        element = MathTex(r"a_{ij} \text{ = element in row } i \text{, column } j", font_size=24)
        element.to_edge(DOWN, buff=0.8)
        
        self.play(Write(element))
        
        self.wait(2)
```

## 2. Matrix Addition and Subtraction

```python
from manim import *
import numpy as np

class MatrixAddition(Scene):
    def construct(self):
        title = Text("Matrix Addition", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        rule = Text("Matrices must have the same order", font_size=22, color=YELLOW)
        rule.next_to(title, DOWN, buff=0.2)
        self.play(Write(rule))
        
        # Two matrices
        A = Matrix([
            [2, 5],
            [1, 3]
        ], h_buff=0.8)
        A.shift(LEFT * 4)
        A_label = MathTex("A =", font_size=28).next_to(A, LEFT)
        
        plus = MathTex("+", font_size=36)
        plus.next_to(A, RIGHT, buff=0.3)
        
        B = Matrix([
            [1, -2],
            [4, 0]
        ], h_buff=0.8)
        B.next_to(plus, RIGHT, buff=0.3)
        B_label = MathTex("B =", font_size=28).next_to(B, LEFT)
        
        equals = MathTex("=", font_size=36)
        equals.next_to(B, RIGHT, buff=0.3)
        
        self.play(Create(A), Write(A_label))
        self.play(Write(plus), Create(B))
        
        # Result
        C = Matrix([
            [3, 3],
            [5, 3]
        ], h_buff=0.8)
        C.next_to(equals, RIGHT, buff=0.3)
        
        self.play(Write(equals), Create(C))
        
        # Show how each element is calculated
        explanation = VGroup(
            MathTex(r"2 + 1 = 3", font_size=20, color=GREEN),
            MathTex(r"5 + (-2) = 3", font_size=20, color=GREEN),
            MathTex(r"1 + 4 = 5", font_size=20, color=GREEN),
            MathTex(r"3 + 0 = 3", font_size=20, color=GREEN),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.15)
        explanation.to_edge(DOWN, buff=0.5)
        
        self.play(Write(explanation))
        
        self.wait(2)
```

## 4. Matrix Multiplication

```python
from manim import *
import numpy as np

class MatrixMultiplication(Scene):
    def construct(self):
        title = Text("Matrix Multiplication", font_size=34, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        rule = Text("Columns of A must = Rows of B", font_size=20, color=YELLOW)
        rule.next_to(title, DOWN, buff=0.15)
        self.play(Write(rule))
        
        # Order rule
        order = MathTex(r"(m \times n) \cdot (n \times p) = (m \times p)", font_size=22, color=ORANGE)
        order.next_to(rule, DOWN, buff=0.15)
        self.play(Write(order))
        
        # Matrices
        A = Matrix([
            [2, 3],
            [1, 4]
        ], h_buff=0.7)
        A.shift(LEFT * 4 + DOWN * 0.5)
        
        B = Matrix([
            [5, 1],
            [2, 0]
        ], h_buff=0.7)
        B.next_to(A, RIGHT, buff=0.3)
        
        equals = MathTex("=", font_size=30)
        equals.next_to(B, RIGHT, buff=0.3)
        
        self.play(Create(A), Create(B), Write(equals))
        
        # Calculate step by step
        # First element: row 1 of A × col 1 of B
        step1 = MathTex(r"(2)(5) + (3)(2) = 10 + 6 = 16", font_size=18, color=GREEN)
        step1.shift(RIGHT * 2 + UP * 0.5)
        
        step2 = MathTex(r"(2)(1) + (3)(0) = 2 + 0 = 2", font_size=18, color=GREEN)
        step2.next_to(step1, DOWN, buff=0.15)
        
        step3 = MathTex(r"(1)(5) + (4)(2) = 5 + 8 = 13", font_size=18, color=GREEN)
        step3.next_to(step2, DOWN, buff=0.15)
        
        step4 = MathTex(r"(1)(1) + (4)(0) = 1 + 0 = 1", font_size=18, color=GREEN)
        step4.next_to(step3, DOWN, buff=0.15)
        
        self.play(Write(step1))
        self.play(Write(step2))
        self.play(Write(step3))
        self.play(Write(step4))
        
        # Result
        result = Matrix([
            [16, 2],
            [13, 1]
        ], h_buff=0.7)
        result.next_to(equals, RIGHT, buff=0.3)
        
        self.play(Create(result))
        
        self.wait(2)
```

## 8. Solving Simultaneous Equations with Matrices

```python
from manim import *
import numpy as np

class SimultaneousEquations(Scene):
    def construct(self):
        title = Text("Solving with Matrices", font_size=34, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Equations
        eqns = VGroup(
            MathTex(r"2x + 3y = 8", font_size=24),
            MathTex(r"x + 2y = 5", font_size=24),
        ).arrange(DOWN, buff=0.15)
        eqns.next_to(title, DOWN, buff=0.2)
        
        self.play(Write(eqns))
        
        # Matrix form
        matrix_form = MathTex(
            r"\begin{pmatrix} 2 & 3 \\ 1 & 2 \end{pmatrix}"
            r"\begin{pmatrix} x \\ y \end{pmatrix}"
            r"= \begin{pmatrix} 8 \\ 5 \end{pmatrix}",
            font_size=26
        )
        matrix_form.next_to(eqns, DOWN, buff=0.3)
        
        self.play(Write(matrix_form))
        
        # AX = B, so X = A⁻¹B
        method = MathTex(r"AX = B \quad \Rightarrow \quad X = A^{-1}B", font_size=24, color=YELLOW)
        method.next_to(matrix_form, DOWN, buff=0.2)
        
        self.play(Write(method))
        
        # Find inverse
        det = MathTex(r"|A| = (2)(2) - (3)(1) = 4 - 3 = 1", font_size=20)
        det.shift(DOWN * 1)
        
        inverse = MathTex(r"A^{-1} = \begin{pmatrix} 2 & -3 \\ -1 & 2 \end{pmatrix}", font_size=22)
        inverse.next_to(det, DOWN, buff=0.15)
        
        self.play(Write(det), Write(inverse))
        
        # Multiply
        mult = MathTex(
            r"X = \begin{pmatrix} 2 & -3 \\ -1 & 2 \end{pmatrix}"
            r"\begin{pmatrix} 8 \\ 5 \end{pmatrix}"
            r"= \begin{pmatrix} 16-15 \\ -8+10 \end{pmatrix}"
            r"= \begin{pmatrix} 1 \\ 2 \end{pmatrix}",
            font_size=22
        )
        mult.next_to(inverse, DOWN, buff=0.2)
        
        self.play(Write(mult))
        
        # Answer
        answer = MathTex(r"x = 1, \quad y = 2", font_size=28, color=GREEN)
        answer.to_edge(DOWN, buff=0.3)
        
        box = SurroundingRectangle(answer, color=GREEN)
        
        self.play(Write(answer), Create(box))
        
        self.wait(2)
```
