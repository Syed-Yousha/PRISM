## 1. Vector Basics - Notation and Representation

```python
from manim import *
import numpy as np

class VectorBasics(Scene):
    def construct(self):
        title = Text("Vector Notation", font_size=40, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Different notations
        notations = VGroup(
            VGroup(
                Text("Column vector:", font_size=20),
                MathTex(r"\vec{a} = \begin{pmatrix} 3 \\ 4 \end{pmatrix}", font_size=28)
            ).arrange(RIGHT, buff=0.5),
            VGroup(
                Text("Directed line segment:", font_size=20),
                MathTex(r"\overrightarrow{AB}", font_size=28)
            ).arrange(RIGHT, buff=0.5),
            VGroup(
                Text("Bold letter:", font_size=20),
                MathTex(r"\mathbf{a}", font_size=28)
            ).arrange(RIGHT, buff=0.5),
            VGroup(
                Text("Underlined (handwritten):", font_size=20),
                MathTex(r"\underline{a}", font_size=28)
            ).arrange(RIGHT, buff=0.5),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.3)
        notations.shift(LEFT * 1.5)
        
        for n in notations:
            self.play(Write(n), run_time=0.5)
        
        # Visual representation
        axes = Axes(
            x_range=[-1, 5, 1],
            y_range=[-1, 5, 1],
            x_length=3,
            y_length=3,
            axis_config={"include_tip": True}
        )
        axes.shift(RIGHT * 3.5 + DOWN * 0.5)
        
        vector = Arrow(axes.c2p(0, 0), axes.c2p(3, 4), color=YELLOW, buff=0)
        vec_label = MathTex(r"\begin{pmatrix} 3 \\ 4 \end{pmatrix}", font_size=22, color=YELLOW)
        vec_label.next_to(vector.get_center(), RIGHT, buff=0.2)
        
        self.play(Create(axes))
        self.play(Create(vector), Write(vec_label))
        
        # Components
        x_comp = MathTex(r"x = 3", font_size=18, color=RED)
        x_comp.next_to(axes.c2p(1.5, 0), DOWN, buff=0.2)
        
        y_comp = MathTex(r"y = 4", font_size=18, color=GREEN)
        y_comp.next_to(axes.c2p(3, 2), RIGHT, buff=0.2)
        
        self.play(Write(x_comp), Write(y_comp))
        
        self.wait(2)
```

        self.play(Write(title))
        
        # Problem
        problem = Text("P divides AB in ratio 2:3. Find position vector of P.", font_size=18)
        problem.next_to(title, DOWN, buff=0.2)
        
        self.play(Write(problem))
        
        # Given
        given = MathTex(r"\vec{a} = \begin{pmatrix} 1 \\ 4 \end{pmatrix}, \quad \vec{b} = \begin{pmatrix} 6 \\ 9 \end{pmatrix}", font_size=22)
        given.next_to(problem, DOWN, buff=0.2)
        
        self.play(Write(given))
        
        # Solution
        solution = VGroup(
            MathTex(r"\overrightarrow{AP} = \frac{2}{5}\overrightarrow{AB}", font_size=22),
            MathTex(r"\overrightarrow{AB} = \vec{b} - \vec{a} = \begin{pmatrix} 5 \\ 5 \end{pmatrix}", font_size=22),
            MathTex(r"\overrightarrow{AP} = \frac{2}{5}\begin{pmatrix} 5 \\ 5 \end{pmatrix} = \begin{pmatrix} 2 \\ 2 \end{pmatrix}", font_size=22),
            MathTex(r"\vec{p} = \vec{a} + \overrightarrow{AP} = \begin{pmatrix} 1 \\ 4 \end{pmatrix} + \begin{pmatrix} 2 \\ 2 \end{pmatrix} = \begin{pmatrix} 3 \\ 6 \end{pmatrix}", font_size=22, color=GREEN),
        ).arrange(DOWN, buff=0.15)
        solution.shift(DOWN * 1)
        
        for s in solution:
            self.play(Write(s), run_time=0.6)
        
        # Formula box
        formula = MathTex(r"\text{Ratio } m:n \Rightarrow \vec{p} = \frac{n\vec{a} + m\vec{b}}{m+n}", font_size=22, color=ORANGE)
        formula.to_edge(DOWN, buff=0.3)
        
        box = SurroundingRectangle(formula, color=ORANGE)
        
        self.play(Write(formula), Create(box))
        
        self.wait(2)
