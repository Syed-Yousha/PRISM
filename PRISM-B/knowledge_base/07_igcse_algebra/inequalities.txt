## INEQUALITY SYMBOLS
```python
# Understanding symbols
title = Text("Inequality Symbols", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Symbols table
symbols = VGroup(
    VGroup(
        MathTex(">", font_size=56, color=GREEN),
        Text("greater than", font_size=24),
        MathTex("5 > 3", font_size=28, color=GRAY)
    ).arrange(RIGHT, buff=0.5),
    
    VGroup(
        MathTex("<", font_size=56, color=RED),
        Text("less than", font_size=24),
        MathTex("2 < 7", font_size=28, color=GRAY)
    ).arrange(RIGHT, buff=0.5),
    
    VGroup(
        MathTex("\\geq", font_size=56, color=BLUE),
        Text("greater than or equal to", font_size=24),
        MathTex("x \\geq 5", font_size=28, color=GRAY)
    ).arrange(RIGHT, buff=0.3),
    
    VGroup(
        MathTex("\\leq", font_size=56, color=ORANGE),
        Text("less than or equal to", font_size=24),
        MathTex("y \\leq 10", font_size=28, color=GRAY)
    ).arrange(RIGHT, buff=0.3)
).arrange(DOWN, buff=0.5, aligned_edge=LEFT)
symbols.move_to(ORIGIN)

# Memory tip
tip = VGroup(
    Rectangle(width=9, height=0.8, color=YELLOW, fill_opacity=0.2),
    Text("Tip: The symbol opens towards the BIGGER number!", font_size=22, color=YELLOW)
)
tip[1].move_to(tip[0])
tip.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
for symbol in symbols:
    self.play(Write(symbol), run_time=0.6)
self.play(Create(tip), run_time=1.0)
```

## NUMBER LINE REPRESENTATION

## SOLVING LINEAR INEQUALITIES
```python
# Solve like equations BUT reverse sign when × or ÷ by negative
title = Text("Solving Linear Inequalities", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Example 1: Normal case
ex1 = VGroup(
    Text("Example 1:", font_size=22, color=BLUE),
    MathTex("3x + 5 > 14", font_size=36),
    MathTex("3x > 9", font_size=36),
    MathTex("x > 3", font_size=40, color=GREEN)
).arrange(DOWN, buff=0.2)
ex1.shift(UP * 0.8 + LEFT * 3.5)

# Example 2: Dividing by negative
ex2 = VGroup(
    Text("Example 2:", font_size=22, color=RED),
    MathTex("-2x < 6", font_size=36),
    Text("Divide by -2, REVERSE sign!", font_size=18, color=RED),
    MathTex("x > -3", font_size=40, color=GREEN)
).arrange(DOWN, buff=0.2)
ex2.shift(UP * 0.8 + RIGHT * 3)

# Key rule
rule = VGroup(
    Rectangle(width=10, height=1.2, color=RED, fill_opacity=0.2),
    VGroup(
        Text("IMPORTANT: When multiplying or dividing", font_size=22, color=RED),
        Text("by a NEGATIVE number, REVERSE the inequality!", font_size=22, color=RED)
    ).arrange(DOWN, buff=0.1)
)
rule[1].move_to(rule[0])
rule.shift(DOWN * 1.5)

# Visual
nl = NumberLine(x_range=[-5, 5, 1], length=8, include_numbers=True).shift(DOWN * 2.8)
dot = Circle(radius=0.12, color=GREEN, fill_opacity=0).move_to(nl.n2p(-3))
arrow = Arrow(nl.n2p(-3), nl.n2p(5), color=GREEN, buff=0.12)

self.play(Write(title), run_time=0.8)
self.play(Write(ex1), run_time=2.0)
self.play(Write(ex2), run_time=2.0)
self.play(Create(rule), run_time=1.5)
self.play(Create(nl), Create(dot), Create(arrow), run_time=1.0)
```

## COMPOUND INEQUALITIES
```python
# Double inequalities like 2 < x ≤ 5
title = Text("Compound Inequalities", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Example
example = MathTex("-1 < x \\leq 4", font_size=48, color=WHITE)
example.shift(UP * 1.5)

# Meaning
meaning = VGroup(
    Text("This means:", font_size=24, color=GRAY),
    MathTex("x > -1 \\text{ AND } x \\leq 4", font_size=32)
).arrange(DOWN, buff=0.2)
meaning.shift(UP * 0.5)

# Number line
nl = NumberLine(x_range=[-3, 6, 1], length=9, include_numbers=True).shift(DOWN * 0.5)
dot_left = Circle(radius=0.12, color=GREEN, fill_opacity=0).move_to(nl.n2p(-1))
dot_right = Dot(nl.n2p(4), color=GREEN, radius=0.12)
line = Line(nl.n2p(-1), nl.n2p(4), color=GREEN, stroke_width=6)
label = Text("Open at -1, Closed at 4", font_size=20, color=GREEN).shift(DOWN * 1.2)

# Solving compound
solving = VGroup(
    Text("Solving compound inequalities:", font_size=22, color=ORANGE),
    MathTex("3 \\leq 2x + 1 < 9", font_size=32),
    MathTex("2 \\leq 2x < 8", font_size=32, color=BLUE),
    Text("(subtract 1 from all parts)", font_size=18, color=GRAY),
    MathTex("1 \\leq x < 4", font_size=36, color=GREEN),
    Text("(divide all parts by 2)", font_size=18, color=GRAY)
).arrange(DOWN, buff=0.15)
solving.shift(DOWN * 2.5)

self.play(Write(title), run_time=0.8)
self.play(Write(example), run_time=1.0)
self.play(Write(meaning), run_time=1.0)
self.play(Create(nl), run_time=0.8)
self.play(Create(dot_left), Create(dot_right), Create(line), Write(label), run_time=1.5)
self.play(Write(solving), run_time=2.5)
```

## GRAPHING LINEAR INEQUALITIES (2D)
```python
# Shading regions
title = Text("Graphing Inequalities", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Axes
axes = Axes(
    x_range=[-1, 5, 1],
    y_range=[-1, 5, 1],
    x_length=5,
    y_length=5,
    axis_config={"include_tip": True, "include_numbers": True}
).shift(LEFT * 2.5 + DOWN * 0.3)

# Line y = x + 1
line = axes.plot(lambda x: x + 1, x_range=[-1, 4], color=BLUE)
label = MathTex("y = x + 1", font_size=24, color=BLUE)
label.next_to(line, UP + RIGHT, buff=0.1)

# Shaded region for y < x + 1 (below line)
# FIXED: Correct polygon vertices for region below line y = x + 1
# The line passes through (-1, 0) and (4, 5), we shade below to axis limits
region = Polygon(
    axes.c2p(-1, -1),  # bottom-left corner of visible area
    axes.c2p(4, -1),   # bottom-right corner
    axes.c2p(4, 5),    # top-right (on line: y = 4+1 = 5)
    axes.c2p(-1, 0),   # top-left (on line: y = -1+1 = 0)
    fill_color=GREEN, fill_opacity=0.3, stroke_width=0
)

# Instruction
instr = VGroup(
    MathTex("y < x + 1", font_size=36, color=GREEN),
    Text("Shade BELOW the line", font_size=22, color=GREEN),
    Text("(dashed line = not included)", font_size=18, color=GRAY)
).arrange(DOWN, buff=0.2)
instr.shift(RIGHT * 3 + UP * 0.5)

# Rules
rules = VGroup(
    Text("Rules:", font_size=22, color=YELLOW),
    Text("< or > : dashed line", font_size=18),
    Text("≤ or ≥ : solid line", font_size=18),
    Text("< or ≤ : shade below", font_size=18),
    Text("> or ≥ : shade above", font_size=18)
).arrange(DOWN, buff=0.1, aligned_edge=LEFT)
rules.shift(RIGHT * 3 + DOWN * 1.5)

self.play(Write(title), run_time=0.8)

self.play(Write(instr), run_time=1.5)

```

## IDENTIFYING REGIONS
```python
# Multiple inequalities defining a region
title = Text("Identifying Regions", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Axes
axes = Axes(
    x_range=[-1, 6, 1],
    y_range=[-1, 5, 1],
    x_length=6,
    y_length=5,
    axis_config={"include_tip": True, "include_numbers": True}
).shift(LEFT * 2 + DOWN * 0.3)

# Three boundary lines
# x ≥ 1
line1 = DashedLine(axes.c2p(1, -1), axes.c2p(1, 5), color=BLUE)
# y ≤ 3
line2 = DashedLine(axes.c2p(-1, 3), axes.c2p(6, 3), color=RED)
# y ≥ x - 1
line3 = axes.plot(lambda x: x - 1, x_range=[0, 5], color=GREEN)

# Feasible region (roughly)
region = Polygon(
    axes.c2p(1, 0), axes.c2p(1, 3), axes.c2p(4, 3), axes.c2p(4, 3),
    fill_color=YELLOW, fill_opacity=0.3, stroke_width=0
)

# Labels
constraints = VGroup(
    MathTex("x \\geq 1", font_size=28, color=BLUE),
    MathTex("y \\leq 3", font_size=28, color=RED),
    MathTex("y \\geq x - 1", font_size=28, color=GREEN)
).arrange(DOWN, buff=0.2, aligned_edge=LEFT)
constraints.shift(RIGHT * 3.5 + UP * 1)

# R label in region
R = Text("R", font_size=32, color=YELLOW).move_to(axes.c2p(2.5, 2))

# Method
method = VGroup(
    Text("Find region R:", font_size=20, color=WHITE),
    Text("1. Draw each boundary", font_size=18),
    Text("2. Shade the required side", font_size=18),
    Text("3. Region R satisfies ALL", font_size=18)
).arrange(DOWN, buff=0.1, aligned_edge=LEFT)
method.shift(RIGHT * 3.5 + DOWN * 1.3)

self.play(Write(title), run_time=0.8)
self.play(Create(axes), run_time=1.0)
self.play(Create(line1), Create(line2), Create(line3), run_time=1.5)
self.play(Write(constraints), run_time=1.5)
self.play(FadeIn(region), Write(R), run_time=1.0)
self.play(Write(method), run_time=1.5)
```

## INTEGER SOLUTIONS
```python
# Finding integers in a region
title = Text("Finding Integer Solutions", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Problem
problem = VGroup(
    Text("Find all integers satisfying:", font_size=24, color=WHITE),
    MathTex("-2 \\leq x < 3 \\text{ and } x + y \\leq 4 \\text{ and } y > 0", font_size=28)
).arrange(DOWN, buff=0.2)
problem.shift(UP * 1.5)

# Axes
axes = Axes(
    x_range=[-3, 4, 1],
    y_range=[-1, 5, 1],
    x_length=5,
    y_length=4,
    axis_config={"include_tip": True, "include_numbers": True}
).shift(LEFT * 2.5 + DOWN * 0.8)

# Grid dots for integers
dots = VGroup()
valid_points = []
for x in range(-2, 3):  # -2 ≤ x < 3
    for y in range(1, 5):  # y > 0, and x + y ≤ 4
        if x + y <= 4:
            dot = Dot(axes.c2p(x, y), color=GREEN, radius=0.1)
            dots.add(dot)
            valid_points.append((x, y))

# List of points
points_list = VGroup(
    Text("Integer solutions:", font_size=22, color=GREEN),
    Text(f"{valid_points}", font_size=18, color=WHITE)
).arrange(DOWN, buff=0.2)
points_list.shift(RIGHT * 3 + DOWN * 0.5)

# Count
count = MathTex(f"\\text{{Total: }} {len(valid_points)} \\text{{ points}}", font_size=28, color=ORANGE)
count.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
self.play(Write(problem), run_time=1.5)
self.play(Create(axes), run_time=1.0)
self.play(Create(dots), run_time=1.5)
self.play(Write(points_list), run_time=1.0)
self.play(Write(count), run_time=1.0)
```

## SOLVING QUADRATIC INEQUALITIES
```python
# x² - 5x + 6 < 0
title = Text("Quadratic Inequalities", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Inequality
ineq = MathTex("x^2 - 5x + 6 < 0", font_size=44).shift(UP * 1.8)

# Step 1: Factorize
step1 = VGroup(
    Text("Step 1: Factorize", font_size=22, color=BLUE),
    MathTex("(x - 2)(x - 3) < 0", font_size=36)
).arrange(DOWN, buff=0.15)
step1.shift(UP * 0.8)

# Step 2: Find roots
step2 = VGroup(
    Text("Step 2: Roots are x = 2 and x = 3", font_size=22, color=GREEN)
)
step2.shift(UP * 0.1)

# Step 3: Sketch parabola
axes = Axes(
    x_range=[0, 5, 1],
    y_range=[-2, 4, 1],
    x_length=4,
    y_length=3,
    axis_config={"include_numbers": True}
).shift(LEFT * 2.5 + DOWN * 1.3)

parabola = axes.plot(lambda x: x**2 - 5*x + 6, x_range=[0.5, 4.5], color=BLUE)
root1 = Dot(axes.c2p(2, 0), color=GREEN, radius=0.1)
root2 = Dot(axes.c2p(3, 0), color=GREEN, radius=0.1)

# Shade negative region
shade = axes.get_area(parabola, x_range=[2, 3], color=RED, opacity=0.3)

# Answer
answer = VGroup(
    Text("Below x-axis when:", font_size=20, color=WHITE),
    MathTex("2 < x < 3", font_size=40, color=GREEN)
).arrange(DOWN, buff=0.2)
answer.shift(RIGHT * 3 + DOWN * 1)

# Number line
nl = NumberLine(x_range=[0, 5, 1], length=4, include_numbers=True).shift(DOWN * 3)
line_shade = Line(nl.n2p(2), nl.n2p(3), color=GREEN, stroke_width=6)
circ1 = Circle(radius=0.1, color=GREEN).move_to(nl.n2p(2))
circ2 = Circle(radius=0.1, color=GREEN).move_to(nl.n2p(3))

self.play(Write(title), run_time=0.8)
self.play(Write(ineq), run_time=1.0)
self.play(Write(step1), run_time=1.5)
self.play(Write(step2), run_time=1.0)
self.play(Create(axes), Create(parabola), run_time=1.0)
self.play(Create(root1), Create(root2), FadeIn(shade), run_time=1.0)
self.play(Write(answer), run_time=1.0)
self.play(Create(nl), Create(line_shade), Create(circ1), Create(circ2), run_time=1.0)
```
