## LIKE TERMS VISUALIZATION
```python
# Show like and unlike terms
title = Text("Like and Unlike Terms", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Like terms (same variable and power)
like_title = Text("Like Terms - CAN combine", font_size=24, color=GREEN).shift(UP * 1.3 + LEFT * 3.5)
like_terms = VGroup(
    MathTex("3x + 5x = 8x", font_size=36, color=GREEN),
    MathTex("2a^2 - 4a^2 = -2a^2", font_size=36, color=GREEN),
    MathTex("7xy + 2xy = 9xy", font_size=36, color=GREEN)
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
like_terms.shift(UP * 0 + LEFT * 3)

# Unlike terms (different variable or power)
unlike_title = Text("Unlike Terms - CANNOT combine", font_size=24, color=RED).shift(UP * 1.3 + RIGHT * 2.5)
unlike_terms = VGroup(
    MathTex("3x + 5y", font_size=36, color=RED),
    MathTex("2a + 4a^2", font_size=36, color=RED),
    MathTex("xy + x", font_size=36, color=RED)
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
unlike_terms.shift(UP * 0 + RIGHT * 3)

# Visual boxes
like_box = SurroundingRectangle(like_terms, color=GREEN, buff=0.3)
unlike_box = SurroundingRectangle(unlike_terms, color=RED, buff=0.3)

# Key rule
rule = VGroup(
    Rectangle(width=10, height=0.8, color=YELLOW, fill_opacity=0.2),
    Text("Like terms: same letter(s) with same power(s)", font_size=22, color=YELLOW)
)
rule[1].move_to(rule[0])
rule.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
self.play(Write(like_title), Write(unlike_title), run_time=0.8)
self.play(Write(like_terms), Create(like_box), run_time=1.5)
self.play(Write(unlike_terms), Create(unlike_box), run_time=1.5)
self.play(Create(rule), run_time=1.0)
```

## COLLECTING LIKE TERMS
```python
# Step by step simplification
title = Text("Collecting Like Terms", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Expression
expr = MathTex("3x + 2y - 5x + 4 + 3y - 2", font_size=40)
expr.shift(UP * 1.5)

# Color coding
color_coded = MathTex("3x", "-5x", "+", "2y", "+3y", "+", "4", "-2", font_size=40)
color_coded.shift(UP * 0.5)
# x terms in blue, y terms in green, constants in orange
color_coded[0].set_color(BLUE)
color_coded[1].set_color(BLUE)
color_coded[3].set_color(GREEN)
color_coded[4].set_color(GREEN)
color_coded[6].set_color(ORANGE)
color_coded[7].set_color(ORANGE)

# Grouped
grouped = VGroup(
    MathTex("x\\text{-terms: } 3x - 5x = -2x", font_size=32, color=BLUE),
    MathTex("y\\text{-terms: } 2y + 3y = 5y", font_size=32, color=GREEN),
    MathTex("\\text{constants: } 4 - 2 = 2", font_size=32, color=ORANGE)
).arrange(DOWN, buff=0.3, aligned_edge=LEFT)
grouped.shift(DOWN * 0.5)

# Final answer
answer = MathTex("= -2x + 5y + 2", font_size=48, color=WHITE)
answer.to_edge(DOWN, buff=1)
answer_box = SurroundingRectangle(answer, color=GREEN, buff=0.2)

self.play(Write(title), run_time=0.8)
self.play(Write(expr), run_time=1.0)
self.play(Transform(expr, color_coded), run_time=1.5)
self.play(Write(grouped), run_time=2.0)
self.play(Write(answer), Create(answer_box), run_time=1.0)
```

## EXPANDING SINGLE BRACKET
```python
# a(b + c) = ab + ac
title = Text("Expanding Single Brackets", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Rule with visual
rule = MathTex("a(b + c) = ab + ac", font_size=44, color=WHITE)
rule.shift(UP * 1.5)

# Arrow diagram
arrow_diagram = VGroup(
    MathTex("3(x + 4)", font_size=48),
)
arrow_diagram.shift(UP * 0.3)

# Arrows showing distribution
arrow1 = CurvedArrow(
    start_point=UP * 0.5 + LEFT * 1.2,
    end_point=DOWN * 0.2 + LEFT * 0.3,
    color=BLUE,
    angle=-TAU/4
)
arrow2 = CurvedArrow(
    start_point=UP * 0.5 + LEFT * 1.2,
    end_point=DOWN * 0.2 + RIGHT * 0.8,
    color=BLUE,
    angle=-TAU/4
)

# Steps
step1 = MathTex("= 3 \\times x + 3 \\times 4", font_size=36).shift(DOWN * 0.5)
step2 = MathTex("= 3x + 12", font_size=44, color=GREEN).shift(DOWN * 1.3)

# More examples
examples = VGroup(
    MathTex("5(2a - 3) = 10a - 15", font_size=32),
    MathTex("-2(x + 5) = -2x - 10", font_size=32),
    MathTex("x(x - 3) = x^2 - 3x", font_size=32)
).arrange(DOWN, buff=0.25, aligned_edge=LEFT)
```

## EXPANDING DOUBLE BRACKETS - FOIL
```python
# (a + b)(c + d)
title = Text("Expanding Double Brackets - FOIL", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

    Text("L", font_size=32, color=BLUE), Text("ast", font_size=24)
).arrange(RIGHT, buff=0.05)

# Example
expr = MathTex("(x + 3)(x + 5)", font_size=44).shift(UP * 0.5)

    MathTex("+5", font_size=32).shift(UP * 0.5 + RIGHT * 0.8),

    Rectangle(width=1.2, height=0.8, color=ORANGE, fill_opacity=0.3).shift(RIGHT * 0.8),
```

## DIFFERENCE OF TWO SQUARES
```python
# (a + b)(a - b) = a² - b²
title = Text("Difference of Two Squares", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Formula
formula = MathTex("(a + b)(a - b) = a^2 - b^2", font_size=48, color=WHITE)
formula.shift(UP * 1.5)
formula_box = SurroundingRectangle(formula, color=YELLOW, buff=0.3)

# Proof
proof = VGroup(
    MathTex("(a + b)(a - b)", font_size=32),
    MathTex("= a^2 - ab + ab - b^2", font_size=32),
    MathTex("= a^2 - b^2", font_size=36, color=GREEN)
).arrange(DOWN, buff=0.25)
proof.shift(DOWN * 0.2)

# Examples
examples = VGroup(
    MathTex("(x + 3)(x - 3) = x^2 - 9", font_size=32),
    MathTex("(2y + 5)(2y - 5) = 4y^2 - 25", font_size=32),
    MathTex("(a + 1)(a - 1) = a^2 - 1", font_size=32)
).arrange(DOWN, buff=0.25, aligned_edge=LEFT)
examples.to_edge(DOWN, buff=0.7)

self.play(Write(title), run_time=0.8)
self.play(Write(formula), Create(formula_box), run_time=1.5)
self.play(Write(proof), run_time=2.0)
self.play(Write(examples), run_time=1.5)
```

## SUBSTITUTION
```python
# Replacing variables with numbers
title = Text("Substitution", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Given values
values = MathTex("\\text{If } a = 3, b = -2, c = 5", font_size=32, color=WHITE)
values.shift(UP * 1.5)

# Example 1
ex1 = VGroup(
    MathTex("2a + 3b", font_size=36),
    MathTex("= 2(3) + 3(-2)", font_size=36, color=BLUE),
    MathTex("= 6 + (-6)", font_size=36),
    MathTex("= 0", font_size=40, color=GREEN)
).arrange(DOWN, buff=0.2)
ex1.shift(LEFT * 3.5 + DOWN * 0.5)

# Example 2 (more complex)
ex2 = VGroup(
    MathTex("a^2 - bc", font_size=36),
    MathTex("= (3)^2 - (-2)(5)", font_size=36, color=BLUE),
    MathTex("= 9 - (-10)", font_size=36),
    MathTex("= 9 + 10 = 19", font_size=40, color=GREEN)
).arrange(DOWN, buff=0.2)
ex2.shift(RIGHT * 3 + DOWN * 0.5)

# Tip
tip = VGroup(
    Rectangle(width=10, height=0.8, color=ORANGE, fill_opacity=0.2),
    Text("Use brackets when substituting negative numbers!", font_size=22, color=ORANGE)
)
tip[1].move_to(tip[0])
tip.to_edge(DOWN, buff=0.5)

self.play(Write(title), run_time=0.8)
self.play(Write(values), run_time=1.0)
self.play(Write(ex1), run_time=2.0)
self.play(Write(ex2), run_time=2.0)
self.play(Create(tip), run_time=1.0)
```

## PERFECT SQUARE EXPANSION
```python
# (a + b)² and (a - b)²
title = Text("Perfect Square Expansion", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Formula 1
formula1 = MathTex("(a + b)^2 = a^2 + 2ab + b^2", font_size=40, color=GREEN)
formula1.shift(UP * 1.3 + LEFT * 3)

example1 = VGroup(
    MathTex("(x + 4)^2", font_size=32),
    MathTex("= x^2 + 2(x)(4) + 4^2", font_size=32),
    MathTex("= x^2 + 8x + 16", font_size=36, color=GREEN)
).arrange(DOWN, buff=0.2)
example1.shift(UP * 0 + LEFT * 3)

# Formula 2
formula2 = MathTex("(a - b)^2 = a^2 - 2ab + b^2", font_size=40, color=RED)
formula2.shift(UP * 1.3 + RIGHT * 3)

example2 = VGroup(
    MathTex("(x - 3)^2", font_size=32),
    MathTex("= x^2 - 2(x)(3) + 3^2", font_size=32),
    MathTex("= x^2 - 6x + 9", font_size=36, color=RED)
).arrange(DOWN, buff=0.2)
example2.shift(UP * 0 + RIGHT * 3)

# Common mistake
mistake = VGroup(
    Text("Common Mistake:", font_size=24, color=ORANGE),
    MathTex("(x + 3)^2 \\neq x^2 + 9", font_size=32, color=RED),
    MathTex("(x + 3)^2 = x^2 + 6x + 9", font_size=32, color=GREEN)
).arrange(DOWN, buff=0.15)
mistake.to_edge(DOWN, buff=0.6)

self.play(Write(title), run_time=0.8)
self.play(Write(formula1), run_time=1.0)
self.play(Write(example1), run_time=1.5)
self.play(Write(formula2), run_time=1.0)
self.play(Write(example2), run_time=1.5)
self.play(Write(mistake), run_time=1.5)
```

## ALGEBRAIC FRACTIONS
```python
# Simplifying algebraic fractions
title = Text("Algebraic Fractions", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Simplifying
simp_title = Text("Simplifying", font_size=24, color=BLUE).shift(UP * 1.5 + LEFT * 4)

    MathTex("\\frac{6x^2}{3x} = \\frac{6}{3} \\times \\frac{x^2}{x} = 2x", font_size=32),
    MathTex("\\frac{12ab}{4b} = 3a", font_size=32)
).arrange(DOWN, buff=0.3)
simp.shift(UP * 0.5 + LEFT * 3)

# Adding (same denominator)
add_title = Text("Adding (same denominator)", font_size=24, color=GREEN).shift(UP * 1.5 + RIGHT * 3)
add = VGroup(
    MathTex("\\frac{3}{x} + \\frac{5}{x} = \\frac{3+5}{x} = \\frac{8}{x}", font_size=32),
).shift(UP * 0.5 + RIGHT * 3)

# Adding (different denominator)
add2_title = Text("Adding (different denominator)", font_size=24, color=ORANGE).shift(DOWN * 0.5)
add2 = VGroup(
    MathTex("\\frac{2}{x} + \\frac{3}{y}", font_size=32),
    MathTex("= \\frac{2y}{xy} + \\frac{3x}{xy}", font_size=32),
    MathTex("= \\frac{2y + 3x}{xy}", font_size=36, color=GREEN)
).arrange(DOWN, buff=0.2)
add2.shift(DOWN * 1.5)

self.play(Write(title), run_time=0.8)
self.play(Write(simp_title), Write(simp), run_time=1.5)

```
