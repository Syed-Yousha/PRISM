## LABELING RIGHT TRIANGLES
```python
# Opposite, Adjacent, Hypotenuse
title = Text("Labeling a Right Triangle", font_size=36, color=YELLOW).to_edge(UP, buff=0.5)

# Triangle - right angle at bottom-left vertex (LEFT * 2, DOWN * 1.5)
# Vertices: A = bottom-left, B = bottom-right, C = top-left
triangle = Polygon(
    LEFT * 2 + DOWN * 1.5,   # A - right angle here
    RIGHT * 2 + DOWN * 1.5,  # B - bottom right
    LEFT * 2 + UP * 1.5,     # C - top left (where theta is)
    color=WHITE, fill_opacity=0.1
)

# Right angle marker - positioned at vertex A with proper alignment
right_marker = Square(side_length=0.3, color=WHITE, fill_opacity=0)
right_marker.move_to(LEFT * 2 + DOWN * 1.5, aligned_edge=DL)  # FIXED: Align to corner

# Angle theta at vertex C (top-left) - calculate actual angle
# The angle at C is between line CA (going down) and line CB (going down-right)
import numpy as np
angle_at_C = np.arctan(4/3)  # arctan(horizontal/vertical) for this triangle
arc = Arc(radius=0.5, start_angle=-PI/2, angle=-angle_at_C, color=GREEN)  # FIXED: Correct arc
arc.shift(LEFT * 2 + UP * 1.5)
theta = MathTex("\\theta", font_size=32, color=GREEN)
theta.shift(LEFT * 1.3 + UP * 1)

# Labels
hyp = Text("Hypotenuse", font_size=22, color=RED)
hyp.shift(RIGHT * 0.5 + UP * 0.3)
hyp_note = Text("(longest side, opposite right angle)", font_size=14, color=GRAY)
hyp_note.next_to(hyp, DOWN, buff=0.1)

opp = Text("Opposite", font_size=22, color=BLUE)
opp.shift(LEFT * 3.2 + DOWN * 0)
opp_note = Text("(opposite to θ)", font_size=14, color=GRAY)
opp_note.next_to(opp, DOWN, buff=0.1)

adj = Text("Adjacent", font_size=22, color=ORANGE)
adj.shift(DOWN * 2)
adj_note = Text("(next to θ)", font_size=14, color=GRAY)
adj_note.next_to(adj, DOWN, buff=0.1)

# Arrows pointing to the sides
arr_hyp = Arrow(hyp.get_left() + LEFT * 0.2, RIGHT * 0 + DOWN * 0.2, color=RED, buff=0)
arr_opp = Arrow(opp.get_right() + RIGHT * 0.1, LEFT * 2 + DOWN * 0, color=BLUE, buff=0)
arr_adj = Arrow(adj.get_top(), LEFT * 0 + DOWN * 1.5, color=ORANGE, buff=0)

self.play(Write(title), run_time=0.8)
self.play(Create(triangle), Create(right_marker), run_time=1.0)
self.play(Create(arc), Write(theta), run_time=1.0)
self.play(Write(hyp), Write(hyp_note), Create(arr_hyp), run_time=1.0)
self.play(Write(opp), Write(opp_note), Create(arr_opp), run_time=1.0)
self.play(Write(adj), Write(adj_note), Create(arr_adj), run_time=1.0)
```
