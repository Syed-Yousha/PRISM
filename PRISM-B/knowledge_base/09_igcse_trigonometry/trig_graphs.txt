## 1. Graph of y = sin(x)

```python
from manim import *
import numpy as np

class SineGraph(Scene):
    def construct(self):
        title = Text("Graph of y = sin(x)", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Axes
        axes = Axes(
            x_range=[-0.5, 2.5*PI, PI/2],
            y_range=[-1.5, 1.5, 0.5],
            x_length=10,
            y_length=4,
            axis_config={"include_tip": True}
        )
        
        # Custom x labels
        x_labels = VGroup(
            MathTex(r"\frac{\pi}{2}").scale(0.6).next_to(axes.c2p(PI/2, 0), DOWN),
            MathTex(r"\pi").scale(0.6).next_to(axes.c2p(PI, 0), DOWN),
            MathTex(r"\frac{3\pi}{2}").scale(0.6).next_to(axes.c2p(3*PI/2, 0), DOWN),
            MathTex(r"2\pi").scale(0.6).next_to(axes.c2p(2*PI, 0), DOWN),
        )
        
        y_labels = VGroup(
            MathTex("1").scale(0.6).next_to(axes.c2p(0, 1), LEFT),
            MathTex("-1").scale(0.6).next_to(axes.c2p(0, -1), LEFT),
        )
        
        self.play(Create(axes), Write(x_labels), Write(y_labels))
        
        # Sine curve
        sine_curve = axes.plot(lambda x: np.sin(x), x_range=[0, 2*PI], color=RED)
        
        self.play(Create(sine_curve), run_time=2)
        
        # Key features
        features = VGroup(
            Text("• Period = 2π", font_size=20, color=WHITE),
            Text("• Amplitude = 1", font_size=20, color=WHITE),
            Text("• Range: -1 ≤ y ≤ 1", font_size=20, color=WHITE),
            Text("• Passes through origin", font_size=20, color=WHITE),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.15)
        features.to_edge(RIGHT).shift(DOWN * 0.5)
        
        self.play(Write(features))
        
        # Mark key points
        key_points = [
            (0, 0), (PI/2, 1), (PI, 0), (3*PI/2, -1), (2*PI, 0)
        ]
        
        dots = VGroup()
        for x, y in key_points:
            dot = Dot(axes.c2p(x, y), color=YELLOW)
            dots.add(dot)
        
        self.play(Create(dots))
        
        self.wait(2)
```

from manim import *
import numpy as np

class TangentGraph(Scene):
    def construct(self):
        title = Text("Graph of y = tan(x)", font_size=36, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        # Axes
        axes = Axes(
            x_range=[-PI, PI, PI/4],
            y_range=[-4, 4, 1],
            x_length=10,
            y_length=5,
            axis_config={"include_tip": True}
        )
        
        # Custom labels
        x_labels = VGroup(
            MathTex(r"-\frac{\pi}{2}").scale(0.5).next_to(axes.c2p(-PI/2, 0), DOWN),
            MathTex(r"\frac{\pi}{2}").scale(0.5).next_to(axes.c2p(PI/2, 0), DOWN),
        )
        
        self.play(Create(axes), Write(x_labels))
        
        # Asymptotes
        asymp1 = DashedLine(axes.c2p(-PI/2, -4), axes.c2p(-PI/2, 4), color=RED)
        asymp2 = DashedLine(axes.c2p(PI/2, -4), axes.c2p(PI/2, 4), color=RED)
        
        asymp_label = Text("Asymptotes", font_size=18, color=RED)
        asymp_label.next_to(asymp2, RIGHT)
        
        self.play(Create(asymp1), Create(asymp2), Write(asymp_label))
        
        # Tangent curves (avoiding asymptotes)
        tan_curve1 = axes.plot(
            lambda x: np.tan(x), 
            x_range=[-PI/2 + 0.1, PI/2 - 0.1],
            color=PURPLE
        )
        
        self.play(Create(tan_curve1), run_time=1.5)
        
        # Key features
        features = VGroup(
            Text("• Period = π", font_size=20, color=WHITE),
            Text("• No amplitude", font_size=20, color=WHITE),
            Text("• Range: all real numbers", font_size=20, color=WHITE),
            Text("• Asymptotes at ±π/2, ±3π/2, ...", font_size=18, color=WHITE),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.15)
        features.to_edge(RIGHT).shift(DOWN * 1)
        
        self.play(Write(features))
        
        # Mark origin
        origin_dot = Dot(axes.c2p(0, 0), color=YELLOW)
        self.play(Create(origin_dot))
        
        self.wait(2)
```

        self.play(Create(sin_half_x_curve), Write(sin_half_x_label), Write(period_half))

        formula.to_edge(DOWN, buff=0.5)
        box = SurroundingRectangle(formula, color=YELLOW)
        
        self.play(Write(formula), Create(box))
        
        self.wait(2)

            y_range=[-2, 4, 1],
            x_length=6,
            y_length=4,
            axis_config={"include_tip": True}
        )
        axes.shift(RIGHT * 2)
        
        self.play(Create(axes))
        
        # Example: y = 2sin(x) + 1
        example_label = MathTex(r"y = 2\sin(x) + 1", font_size=24, color=ORANGE)
        example_label.next_to(axes, UP)
        
        self.play(Write(example_label))
        
        # Mark key points
        # Max at (π/2, 3), Min at (3π/2, -1), Zeros, Middle line at y=1
        
        # Middle line
        mid_line = DashedLine(axes.c2p(0, 1), axes.c2p(2*PI, 1), color=YELLOW, stroke_opacity=0.5)
        self.play(Create(mid_line))
        
        # Key points
        key_points = [
            (0, 1, "Start"),
            (PI/2, 3, "Max"),
            (PI, 1, ""),
            (3*PI/2, -1, "Min"),
            (2*PI, 1, "End"),
        ]
        
        dots = VGroup()
        for x, y, label_text in key_points:
            dot = Dot(axes.c2p(x, y), color=YELLOW)
            dots.add(dot)
            if label_text:
                label = Text(label_text, font_size=14, color=YELLOW)
                label.next_to(dot, UP if y >= 1 else DOWN)
                dots.add(label)
        
        self.play(Create(dots))
        
        # Draw the curve
        curve = axes.plot(lambda x: 2*np.sin(x) + 1, x_range=[0, 2*PI], color=RED)
        self.play(Create(curve))
        
        # Summary
        summary = VGroup(
            Text("Amplitude = 2", font_size=16, color=WHITE),
            Text("Period = 2π", font_size=16, color=WHITE),
            Text("Range: [-1, 3]", font_size=16, color=WHITE),
        ).arrange(DOWN, aligned_edge=LEFT, buff=0.1)
        summary.next_to(axes, DOWN)
        
        self.play(Write(summary))
        
        self.wait(2)
```

## 11. CAST Diagram for Reference Angles

from manim import *
import numpy as np

class CASTDiagram(Scene):
    def construct(self):
        title = Text("CAST Diagram", font_size=40, color=BLUE)
        title.to_edge(UP)
        self.play(Write(title))
        
        subtitle = Text("Which trig functions are positive?", font_size=24, color=GRAY)

        self.play(Write(subtitle))
        
        # Axes
        x_axis = Line(LEFT * 3, RIGHT * 3, color=WHITE)
        y_axis = Line(DOWN * 2.5, UP * 2.5, color=WHITE)
        
        self.play(Create(x_axis), Create(y_axis))
        
        # Quadrant labels
        q1 = Text("ALL\n(+)", font_size=28, color=GREEN, weight=BOLD)
        q1.shift(RIGHT * 1.5 + UP * 1.2)
        
        q2 = Text("SIN\n(+)", font_size=28, color=RED, weight=BOLD)
        q2.shift(LEFT * 1.5 + UP * 1.2)
        
        q3 = Text("TAN\n(+)", font_size=28, color=YELLOW, weight=BOLD)
        q3.shift(LEFT * 1.5 + DOWN * 1.2)
        
        q4 = Text("COS\n(+)", font_size=28, color=PURPLE, weight=BOLD)
        q4.shift(RIGHT * 1.5 + DOWN * 1.2)
        
        self.play(Write(q1), Write(q2), Write(q3), Write(q4))
        
        # Angle ranges
        angle_labels = VGroup(
